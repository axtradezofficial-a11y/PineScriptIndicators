f_passesThreshold(_top, _bottom, _atr) =>
    gapSize = _top - _bottom
    bool pass = true
    if i_fvgThresholdMode == "Ticks"
        pass := gapSize >= i_fvgThresholdTicks * syminfo.mintick
    else if i_fvgThresholdMode == "Percent (ATR)"
        pass := gapSize >= i_fvgThresholdTicks * _atr
    pass

f_nearestUntakenBSL(_fromBar) =>
    float result = na
    if array.size(bslArray) > 0
        for i = array.size(bslArray) - 1 to 0
            SwingPoint sp = array.get(bslArray, i)
            if not sp.taken and sp.barIndex < _fromBar
                result := sp.price
                break
    result

f_findExactHighBar(float val) =>
    int targetBar = bar_index
    float minDiff = 1e10
    for i = 0 to 150
        diff = math.abs(high[i] - val)
        if diff < minDiff
            minDiff := diff
            targetBar := bar_index - i
        if diff == 0
            break
    targetBar

f_findExactLowBar(float val) =>
    int targetBar = bar_index
    float minDiff = 1e10
    for i = 0 to 150
        diff = math.abs(low[i] - val)
        if diff < minDiff
            minDiff := diff
            targetBar := bar_index - i
        if diff == 0
            break
    targetBar

f_formatTimeframe(string _tf) =>
    float val = str.tonumber(_tf)
    string result = _tf
    if not na(val)
        if val >= 1440
            result := str.tostring(math.round(val / 1440)) + "D"
        else if val >= 60
            result := str.tostring(math.round(val / 60)) + "H"
        else
            result := str.tostring(math.round(val)) + "m"
    else
        if _tf == "D" or _tf == "1D"
            result := "1D"
        else if _tf == "W" or _tf == "1W"
            result := "1W"
        else if _tf == "M" or _tf == "1M"
            result := "1M"
    result

f_checkInvalidation(IFVG_Setup s) =>
    bool invalid = false
    if s.state == "INVERTED"
        if i_invalidCond == "Close"
            if s.isBullish and close < s.invalidLevel
                invalid := true
            else if not s.isBullish and close > s.invalidLevel
                invalid := true
        else if i_invalidCond == "Swing"
            sweptLevel = na(s.invalidSwingLevel) ? s.invalidLevel : s.invalidSwingLevel
            if s.isBullish and close < sweptLevel
                invalid := true
            else if not s.isBullish and close > sweptLevel
                invalid := true
    invalid

if array.size(activeIFVGs) > 0
    for i = array.size(activeIFVGs) - 1 to 0
        IFVG_Setup s = array.get(activeIFVGs, i)
        if s.state == "INVERTED" and f_checkInvalidation(s)
            s.state           := "INVALIDATED"
            s.invalidBarIndex := bar_index
            if i_alertOnInvalid
                alert((s.isBullish ? "+IFVG" : "-IFVG") + " Invalidated @ " + str.tostring(close), alert.freq_once_per_bar_close)

f_createSetupVisuals(IFVG_Setup s) =>
    bool ghost   = false
    bool invalid = s.state == "INVALIDATED"
    col = f_dirColor(s.isBullish, ghost, invalid)

    boxRight = i_extendBoxes ? bar_index : (s.leftBar + 2)
    boxCol = i_showBoxes ? color.new(col, invalid ? i_invalidTransparency : i_boxTransparency) : color.new(color.black, 100)
    borderColor = i_showBoxes ? color.new(col, i_borderWidth == 0 ? 100 : 40) : color.new(color.black, 100)
    s.visualBox := box.new(left = s.leftBar, top = s.top, right = boxRight, bottom = s.bottom, border_color = borderColor, border_width = i_borderWidth, bgcolor = boxCol, extend = extend.none)

    if i_showCE
        s.ceLine := line.new(x1 = s.leftBar, y1 = s.mid, x2 = boxRight, y2 = s.mid, color = color.new(col, 20), width = 1, style = line.style_dotted)

    entryPrice = na(s.entryPrice) ? (s.isBullish ? s.top : s.bottom) : s.entryPrice
    int limitX = math.min(bar_index, s.inversionBar + 4)

f_updateSetupVisuals(IFVG_Setup s) =>
    bool ghost   = false
    bool invalid = s.state == "INVALIDATED"
    col = f_dirColor(s.isBullish, ghost, invalid)
    boxCol = color.new(col, invalid ? i_invalidTransparency : i_boxTransparency)

    int boxRight = s.leftBar + 2
    if i_extendBoxes
        if s.state == "INVERTED"
            boxRight := bar_index
        else if invalid and not na(s.invalidBarIndex)
            boxRight := s.invalidBarIndex
        else if s.state == "TARGET_HIT" and not na(s.targetBarIndex)
            boxRight := s.targetBarIndex

    if not na(s.visualBox)
        boxCol = i_showBoxes ? color.new(col, invalid ? i_invalidTransparency : i_boxTransparency) : color.new(color.black, 100)
        borderColor = i_showBoxes ? color.new(col, i_borderWidth == 0 ? 100 : 40) : color.new(color.black, 100)
        box.set_right(s.visualBox, boxRight)
        box.set_border_color(s.visualBox, borderColor)
        box.set_bgcolor(s.visualBox, boxCol)

    if i_showCE and not na(s.ceLine)
        line.set_x2(s.ceLine, boxRight)
        line.set_color(s.ceLine, color.new(col, 20))

    if not na(s.entryLine)
        stopX = math.min(bar_index, s.inversionBar + 4)
        line.set_x2(s.entryLine, stopX)
        line.set_color(s.entryLine, col)

    if not na(s.tagLabel)
        int labelX = math.min(bar_index, s.inversionBar + 4)
        if invalid and not na(s.invalidBarIndex)
            labelX := math.min(s.invalidBarIndex, s.inversionBar + 4)
            
        label.set_x(s.tagLabel, labelX)
            
        bool hasWarning = false
        if s.state == "INVERTED" and i_invalidCond == "Swing"
            if s.isBullish and close < s.invalidLevel
                hasWarning := true
            else if not s.isBullish and close > s.invalidLevel
                hasWarning := true

        string tagTxt = (s.isBullish ? "+𝗜𝗙𝗩𝗚" : "-𝗜𝗙𝗩𝗚") + (invalid ? " ✗" : (hasWarning ? " ⚠️" : ""))
        label.set_text(s.tagLabel, tagTxt)
        label.set_textcolor(s.tagLabel, col)

    if i_showTargets and not na(s.targetLine)
        if invalid
            line.set_color(s.targetLine, color.new(color.black, 100))
            if not na(s.targetLabel)
                label.set_textcolor(s.targetLabel, color.new(color.black, 100))
        else
            int stopX = (i_stopAtTarget and s.targetHit) ? s.targetBarIndex : bar_index
            line.set_x2(s.targetLine, stopX)
            line.set_color(s.targetLine, color.new(col, s.targetHit ? 0 : 30))
            if not na(s.targetLabel)
                label.set_x(s.targetLabel, stopX)
                label.set_textcolor(s.targetLabel, col)
                
                if s.targetHit and not na(s.targetBarIndex)
                    offset = bar_index - s.targetBarIndex
                    targetY = s.isBullish ? high[offset] : low[offset]
                    label.set_y(s.targetLabel, targetY)
                    label.set_text(s.targetLabel, "𝗧")
                    label.set_style(s.targetLabel, s.isBullish ? label.style_label_down : label.style_label_up)
                else
                    label.set_y(s.targetLabel, s.targetPrice)
                    label.set_text(s.targetLabel, s.isBullish ? "𝗕𝗦𝗟" : "𝗦𝗦𝗟")
                    label.set_style(s.targetLabel, label.style_label_left)
