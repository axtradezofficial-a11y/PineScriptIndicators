volSma = ta.sma(volume, 20)

// Delta aproximado do candle: soma o volume das velas de um timeframe MENOR que fecharam em
// alta e o das que fecharam em baixa, e devolve o saldo dividido pelo volume total (+1 = tudo
// comprador, -1 = tudo vendedor).
// O TF menor sai do input "Delta: TF intrabar". No Auto ele é 1/6 do TF do gráfico e NUNCA
// desce abaixo de 1 minuto, porque dado de segundos depende do plano do TradingView — assim o
// indicador não quebra em conta nenhuma. Quem tem segundos troca o input para 10S/15S e ganha
// um delta bem mais fino.
// Sem dado intrabar (plano sem segundos, histórico antigo, ativo sem volume) os arrays voltam
// vazios e o resultado é `na` — aí o critério sai da conta da nota em vez de contar como falha.
dlt_auto = int(math.max(60, timeframe.in_seconds() / 6))
dlt_tf = switch deltaTfStr
    "Auto" => dlt_auto < 86400 ? str.tostring(int(dlt_auto / 60)) : "1D"
    "1m"   => "1"
    "5m"   => "5"
    =>        deltaTfStr
[dlt_upArr, dlt_dnArr] = request.security_lower_tf(syminfo.tickerid, dlt_tf, [close > open ? volume : 0.0, close < open ? volume : 0.0])
float dlt_up  = dlt_upArr.size() > 0 ? dlt_upArr.sum() : na
float dlt_dn  = dlt_dnArr.size() > 0 ? dlt_dnArr.sum() : na
float dlt_tot = na(dlt_up) ? na : dlt_up + dlt_dn
float deltaRatio = na(dlt_tot) or dlt_tot <= 0 ? na : (dlt_up - dlt_dn) / dlt_tot

// O candle de confirmação do IFVG é sempre o ANTERIOR (o gatilho usa close[1]), então é o delta
// dele que interessa. Fica guardado aqui fora porque função Pine não pode usar o histórico de

// Liquidez ALVO do setup: o nível ainda não varrido mais próximo na direção do trade.
// Candidatos: os swings do detector de sweep (stHigh/stLow — o findSweep tira dali tudo que
// já foi varrido) e os níveis nomeados PDH/PWH (compra) ou PDL/PWL (venda).
conf_FindTarget(bool bullish, float entry) =>
    float best = na
    SwingPoint[] pts = bullish ? stHigh.points : stLow.points
    if pts.size() > 0
        for i = 0 to pts.size() - 1
            SwingPoint p = pts.get(i)
            float y = p.y
            if (bullish ? y > entry : y < entry) and (na(best) or (bullish ? y < best : y > best))
                best := y
    LiqLevel[] lv = bullish ? liqHighs : liqLows
    if lv.size() > 0
        for i = 0 to lv.size() - 1
            LiqLevel lq = lv.get(i)
            float y2 = lq.price
            if not na(y2) and (bullish ? y2 > entry : y2 < entry) and (na(best) or (bullish ? y2 < best : y2 > best))
                best := y2
    best

// Caminho limpo: nenhum FVG do timeframe maior CONTRÁRIO ao setup (e ainda válido) entre a
// entrada e o alvo. Um FVG de baixa no meio do caminho de uma compra é obstáculo — é ali que
// o preço tende a reagir antes de entregar a liquidez.
conf_PathClear(bool bullish, float entry, float tgt) =>
    bool clear = true
    if not na(tgt)
        HtfFVG[] opp = bullish ? htfFvgBear : htfFvgBull
        float lo = math.min(entry, tgt)
        float hi = math.max(entry, tgt)
        if opp.size() > 0
            for i = 0 to opp.size() - 1
                HtfFVG f = opp.get(i)
                if not f.filled and f.bottom <= hi and f.top >= lo
                    clear := false
    clear

// Barra do ultimo SMT que CONTA como confluencia, por direcao (o mais recente entre os
// detectores habilitados). O pivo FORTE conta sempre; o pavio adjacente so com o input
// smtCountWicks ligado. SMT de FVG e divergencia FRACA nunca entram.
conf_SmtRefBar(bool bullish) =>
    int idx  = bullish ? 0 : 1
    int rp   = array.get(smtRefBars, idx)
    int rw   = array.get(smtWickBars, idx)
    int best = rp
    if smtCountWicks and not na(rw)
        if na(best)
            best := rw
        else if rw > best
            best := rw
    best

// Liquidity Sweep pelo PRECO - segundo caminho do criterio (o primeiro e o de TEMPO, o
// e.swept: "o swing de origem varreu"). Varre os labels de varrida ja finalizados a partir
// do swing e aceita se algum nivel varrido cair na altura da zona do IFVG, com a folga do
// input liqAtTol. Pega o caso de a varrida ter acontecido ANTES do swing do setup.
conf_AtSweptLiq(int fromBar, float zTop, float zBot, float tol) =>
    bool hit = false
    if sweepTags.size() > 0
        for i = 0 to sweepTags.size() - 1
            SweepTag sg = sweepTags.get(i)
            if sg.bar >= fromBar and sg.price <= zTop + tol and sg.price >= zBot - tol
                hit := true
    hit

// HTF PDA Delivery pela GEOMETRIA - segundo caminho do criterio (o primeiro e o TOQUE,
// htfFvgTouchBull/Bear dentro da janela de 3 barras). Aceita se a zona do IFVG SOBREPOE um
// FVG do timeframe maior da MESMA direcao e ainda valido: pega o IFVG que nasce colado na
// borda da zona, em que o toque nao caiu na janela. FVG contrario continua nao contando.
conf_InHtfFvg(bool bullish, float zTop, float zBot) =>
    bool hit = false
    HtfFVG[] same = bullish ? htfFvgBull : htfFvgBear
    if same.size() > 0
        for i = 0 to same.size() - 1
            HtfFVG f = same.get(i)
            if not f.filled and f.bottom <= zTop and f.top >= zBot
                hit := true
    hit

// Nota do setup — CONTAGEM SIMPLES: cada critério vale 1 ponto e a nota sai da fração de
// critérios ATENDIDOS sobre os APLICÁVEIS. Um módulo desligado (SMT / HTF FVG) ou sem dado
// (delta sem intrabar) sai da conta em vez de puxar a nota para baixo.
// O critério IFVG entra sempre atendido: o setup só existe porque houve um IFVG confirmado.
conf_Score(Trade tr) =>
    int hits  = 1
    int total = 1
    hits  += tr.sweptHTF ? 1 : 0
    total += 1
    hits  += tr.clearTgt ? 1 : 0
    total += 1
    hits  += tr.volSpike ? 1 : 0
    total += 1
    if showSMT
        hits  += tr.smtOk ? 1 : 0
        total += 1
    if showHtfFvg
        hits  += tr.htfFvgOk ? 1 : 0
        total += 1
    if not na(tr.deltaVal)
        hits  += tr.deltaImb ? 1 : 0
        total += 1
    [hits, total]

// Letra e cor da nota. 100% = A+ | >=80% = A | >=60% = B | >=40% = C | resto = D.
conf_Grade(int hits, int total) =>
    float pct = total > 0 ? hits / float(total) : 0.0
    string g = pct >= 1 ? "A+" : pct >= 0.8 ? "A" : pct >= 0.6 ? "B" : pct >= 0.4 ? "C" : "D"
    color  c = pct >= 1 ? #00E5B0 : pct >= 0.8 ? color.lime : pct >= 0.6 ? color.yellow : pct >= 0.4 ? color.orange : color.red
    [g, c]

// Guarda um label de liquidez já finalizado no registro, para o anti-sobreposição.
sweep_Archive(label lb, int b, float p, bool named, bool shown) =>
    if not na(lb)
        sweepTags.push(SweepTag.new(lb, b, p, named, shown))
        while sweepTags.size() > 60
            sweepTags.shift()

// Chamado toda vez que o label ATIVO é criado ou reposicionado. Apaga o texto dos vizinhos
// antigos que ficariam colados nele e devolve se o ativo pode mostrar o próprio texto.
// A linha de liquidez nunca é removida — só o texto some, para o gráfico não virar sopa.
sweep_Declutter(int b, float p, bool named) =>
    bool show = true
    if declutterSweep and sweepTags.size() > 0
        float vGap = nz(sweepAtr) * sweepGapAtr
        for t in sweepTags
            if t.shown and math.abs(t.bar - b) <= sweepGapBars and math.abs(t.price - p) <= vGap
                if t.named and not named
                    // Nível nomeado tem preferência: quem cede é o label genérico novo.
                    show := false
                else
                    label.set_text(t.lbl, "")
                    t.shown := false
    show

ict_PushCandidate(array<SwingFVG> arr, int fvgBar1, int fvgBar3, float top, float bottom, int oBar, float oPrice) =>

    arr.push(sw)
    if arr.size() > MAX_TRACKED_SETUPS
        arr.shift()
    sw

ict_ScanExtreme(bool ifvgBullish, int fromBar) =>
    int   span      = bar_index - fromBar
    float bestPrice = na
    int   bestBar   = na
    for j = 0 to span
        float val = ifvgBullish ? low[span - j] : high[span - j]
        if na(bestPrice) or (ifvgBullish ? val < bestPrice : val > bestPrice)
            bestPrice := val
            bestBar   := fromBar + j
    [bestBar, bestPrice]

    for part in str.split(str.replace_all(src, " ", ""), ",")
        v = str.tonumber(part)
        if not na(v)
            out.push(v)
    out

// liqLvls = níveis nomeáveis do lado que está sendo varrido, em ordem de prioridade:
// [PWH, PDH] para varrida de topo, [PWL, PDL] para varrida de fundo.
ict_TrackLine(bool isLow, array<SwingFVG> validArr, bool sweepValid, float sweepPrice, int sweepTime, color sweepLineColor, array<LiqLevel> liqLvls) =>
    var float originPrice  = na
    var int   originBar    = na
    var bool  frozen       = false
    var float frozenPrice  = na
    var int   freezeBar    = na
    var SwingFVG liveEntry    = na
    var int      clusterLast1 = na
    var line     sweepLn       = na
    var label    sweepLbl      = na
    var float    sweepLnPrice  = na
    var int      sweepLnTime   = na
    var int      sweepHitTime  = na
    var int      sweepHitBar   = na
    var string   sweepLnTxt    = na
    var bool     sweepNamed    = false
    var bool     sweepShown    = true

    bool  pivotOrigin = isLow ? ta.lowestbars(3) == -1  : ta.highestbars(3) == -1
    bool  pivotOpp    = isLow ? ta.highestbars(3) == -1 : ta.lowestbars(3) == -1
    float originSrc   = isLow ? low[1]  : high[1]
    float oppSrc      = isLow ? high[1] : low[1]
    bool  fvgHere     = isLow ? low[1] > high[3] : high[1] < low[3]
    float fvgTop      = isLow ? low[1] : low[3]
    float fvgBottom   = isLow ? high[3] : high[1]

    if pivotOrigin
        originPrice  := originSrc
        originBar    := bar_index - 1
        frozen       := false
        frozenPrice  := na
        freezeBar    := na
        liveEntry    := na
        clusterLast1 := na
        // O label do swing que acabou não se move mais: vai para o registro, onde passa a
        // contar como vizinho para o anti-sobreposição dos próximos.
        sweep_Archive(sweepLbl, sweepHitBar, sweepLnPrice, sweepNamed, sweepShown)
        sweepLn      := na
        sweepLbl     := na
        sweepLnPrice := na
        sweepLnTime  := na
        sweepHitTime := na
        sweepHitBar  := na
        sweepLnTxt   := na
        sweepNamed   := false
        sweepShown   := true

    if not na(originBar)
        if frozen
            bool broken = isLow ? oppSrc > frozenPrice : oppSrc < frozenPrice
            if broken
                frozen := false

        bool checkFvg = (not frozen or bar_index == freezeBar + 1) and bar_index - 3 >= originBar - 1 and fvgHere

        if checkFvg
            int fvgBar1 = bar_index - 1
            int fvgBar3 = bar_index - 3

            if not na(liveEntry) and not liveEntry.ifvgConfirmed and validArr.indexof(liveEntry) < 0
                liveEntry    := na
                clusterLast1 := na

            if na(liveEntry)
                liveEntry    := ict_PushCandidate(validArr, fvgBar1, fvgBar3, fvgTop, fvgBottom, originBar, originPrice)
                clusterLast1 := fvgBar1
            else if not liveEntry.ifvgConfirmed
                if ifvgMode == "Series" and fvgBar1 == clusterLast1 + 1
                    liveEntry.fvgTop    := math.max(liveEntry.fvgTop, fvgTop)
                    liveEntry.fvgBottom := math.min(liveEntry.fvgBottom, fvgBottom)
                    liveEntry.fvgBar1   := fvgBar1
                    clusterLast1 := fvgBar1
                else
                    bool better = isLow ? fvgTop > liveEntry.fvgTop : fvgBottom < liveEntry.fvgBottom
                    if better
                        int oldIdx = validArr.indexof(liveEntry)
                        if oldIdx >= 0
                            validArr.remove(oldIdx)
                        liveEntry    := ict_PushCandidate(validArr, fvgBar1, fvgBar3, fvgTop, fvgBottom, originBar, originPrice)
                        clusterLast1 := fvgBar1

        // A linha de liquidez vai do pivô que criou o nível (sweepLnTime) até a barra que o
        // varreu (sweepHitTime) e PARA ali — não acompanha mais o preço.
        //
        // O label é ancorado por PIXEL a partir de uma coordenada fixa (barra + preço), nunca
        // por deslocamento de preço — assim ele fica colado na linha e não escorrega no zoom.
        if sweepValid
            bool betterSweep = na(sweepLnPrice) or (isLow ? sweepPrice > sweepLnPrice : sweepPrice < sweepLnPrice)
            if betterSweep
                sweepLnPrice := sweepPrice
                sweepLnTime  := sweepTime
                sweepHitTime := time
                sweepHitBar  := bar_index
                sweepLnTxt   := sweepLabelText
                sweepNamed   := false
                if namePDLevels
                    float pdTol = syminfo.mintick * pdTolTicks
                    for lv in liqLvls
                        if not lv.used and not na(lv.price) and math.abs(sweepPrice - lv.price) <= pdTol
                            sweepLnTxt := lv.name
                            sweepNamed := true
                            lv.used    := true
                            break
                sweepShown := sweep_Declutter(sweepHitBar, sweepLnPrice, sweepNamed)
            // Ancoragem por CANTO, como o PO3 faz nos PDH/PDL e nas BSL/SSL: o canto inferior
            // direito no ponto (label_lower_right) joga o texto para CIMA e para a ESQUERDA;
            // o canto superior direito (label_upper_right) joga para BAIXO e para a esquerda.
            // Assim o texto nunca invade o espaço à direita da ponta da linha, que é onde o
            // preço está — diferente dos estilos com seta, que centram o texto na âncora.
            // Aqui isLow = true significa varrida de TOPO (o pivô de origem é um fundo).
            int sweepLblBar = sweepHitBar - SWEEP_LABEL_BACK
            if na(sweepLn)
                sweepLn  := line.new(sweepLnTime, sweepLnPrice, sweepHitTime, sweepLnPrice, xloc = xloc.bar_time, color = sweepLineColor, style = sweepLineStyle, width = sweepWidth)
                sweepLbl := label.new(sweepLblBar, sweepLnPrice, sweepShown ? sweepLnTxt : "", xloc = xloc.bar_index, style = isLow ? label.style_label_lower_right : label.style_label_upper_right, color = na, textcolor = sweepLineColor, size = sweepLabelSize, text_font_family = font.family_monospace)
            else
                line.set_xy1(sweepLn, sweepLnTime, sweepLnPrice)
                line.set_xy2(sweepLn, sweepHitTime, sweepLnPrice)
                label.set_xy(sweepLbl, sweepLblBar, sweepLnPrice)
                label.set_text(sweepLbl, sweepShown ? sweepLnTxt : "")

        bool swingSwept = not na(sweepLn)

        if not na(liveEntry)
            liveEntry.swept := swingSwept

        if not frozen and pivotOpp
            frozen      := true
            frozenPrice := oppSrc
            freezeBar   := bar_index

// SMT TARDIO - corrige o congelamento do tr.smtOk.
// O SMT de pivo so e DETECTADO smt_L1/L2/L3 barras DEPOIS do pivo (e o que o ta.pivothigh
// exige para confirmar), mas o tr.smtOk e avaliado UMA vez, no candle em que o IFVG
// confirma. Quando o IFVG confirma dentro dessa janela - o caso normal - o sinal ainda nao
// existe e o X ficava gravado para sempre, mesmo com o label ja desenhado no topo.
// Esta funcao roda a cada barra e vira o X em V quando aparece um pivo FORTE cujo PIVO caia
// entre swingBar - smt_L1 e a barra de entrada. SMT posterior ao setup nao conta.
// NAO pode viver dentro do ict_ManageTrades: com "Congelar apos atingir criterio" o setup
// sai do array de trades em 1-2 barras e nunca mais passaria por la.
ict_SmtLate(array<Trade> arr) =>
    if showSMT and arr.size() > 0
        for i = 0 to arr.size() - 1
            Trade t = arr.get(i)
            if not na(t)
                // Campos copiados para locais: `and` nao garante curto-circuito no Pine.
                bool pend = not t.smtOk
                bool live = not t.stopped
                bool hasS = not na(t.swingBar)
                if pend and live and hasS
                    int refBar = conf_SmtRefBar(t.bullish)
                    if not na(refBar) and refBar >= t.swingBar - smt_L1 and refBar <= t.entryBar
                        t.smtOk := true
    true

ict_CheckIFVG(array<SwingFVG> arr, array<Trade> trades, bool sourceBearish, int graceCandles) =>
    int n = arr.size()
    if n > 0
        for i = n - 1 to 0
            SwingFVG e = arr.get(i)
            if not e.ifvgConfirmed
                bool trigger = sourceBearish ? close[1] > e.fvgTop : close[1] < e.fvgBottom
                if trigger
                    e.ifvgConfirmed := true
                    bool  ifvgBullish = sourceBearish
                    float lvl         = close[1]
                    bool  beBreached  = requireBEIntact and (ifvgBullish ? lvl >= e.swingPrice : lvl <= e.swingPrice)
                    bool  passSweepFilter   = not onlyShowSweptIFVG or e.swept

                    if not beBreached and passSweepFilter
                        color lc       = ifvgBullish ? colorIFVGBull : colorIFVGBear
                        color boxColor = ifvgBullish ? colorIFVGBullBox : colorIFVGBearBox
                        e.fvgBox := box.new(e.fvgBar3, e.fvgTop, bar_index, e.fvgBottom, border_color = noColor, bgcolor = boxColor)
                        int    x1  = bar_index - 1
                        int    x2  = x1 + 9
                        bool   volAbove = volume[1] > volSma[1]
                        string txt = ifvgLineText + (ifvgBullish ? "+" : "-") + (volAbove ? " ● " : "")
                        line  entryLine  = line.new(x1, lvl, x2, lvl, color = lc)
                        label entryLabel = label.new(x2, lvl, txt, style = label.style_label_left, color = noColor, textcolor = lc, size = textSize, text_font_family = font.family_monospace)

                        Trade tr = Trade.new(ifvgBullish, e.fvgBox, entryLine, entryLabel, x1, lvl)
                        tr.swingBar := e.swingBar
                        // Liquidity Sweep tem DOIS caminhos: o de TEMPO (o swing de origem varreu)
                        // e o de PRECO (a zona do IFVG nasceu na altura de um nivel ja varrido).
                        tr.sweptHTF := e.swept or conf_AtSweptLiq(e.swingBar, e.fvgTop, e.fvgBottom, liqAtTol * sweepAtr)
                        tr.volSpike := volAbove

                        // Confluências extras, avaliadas no momento da confirmação do IFVG e congeladas no setup:
                        // SMT     -> houve divergência SMT na direção do setup a partir do swing de origem.
                        // HTF FVG -> o preço interagiu com um FVG do timeframe maior durante a formação do setup.
                        int smtRefBar = conf_SmtRefBar(ifvgBullish)
                        tr.smtOk := showSMT and not na(smtRefBar) and smtRefBar >= e.swingBar - smt_L1
                        int htfFvgRefBar = ifvgBullish ? htfFvgTouchBull : htfFvgTouchBear
                        // Tambem dois caminhos: o TOQUE (o preco esteve dentro do FVG HTF durante a
                        // formacao) e a GEOMETRIA (a zona do IFVG sobrepoe um FVG HTF da mesma
                        // direcao ainda valido) - pega o IFVG que nasce colado na borda.
                        tr.htfFvgOk := showHtfFvg and ((not na(htfFvgRefBar) and htfFvgRefBar >= e.swingBar - 3) or conf_InHtfFvg(ifvgBullish, e.fvgTop, e.fvgBottom))

                        // Nível de invalidação (fundo/topo do range do IFVG). É SEMPRE calculado,
                        // independente do toggle "Stoploss" (showSL), pois o ict_ManageTrades depende
                        // dele para apagar o setup quando esse nível é violado.
                        [bestBar, bestPrice] = ict_ScanExtreme(ifvgBullish, e.fvgBar1)
                        int   offset        = bar_index - bestBar
                        bool  candleBullish = close[offset] > open[offset]
                        float bodyPrice     = (ifvgBullish == candleBullish) ? open[offset] : close[offset]
                        float slPrice = switch slPlacement
                            "Wick" => bestPrice
                            "Body" => bodyPrice
                            "IFVG" => ifvgBullish ? e.fvgBottom : e.fvgTop
                        tr.slPrice := slPrice

                        // Clear Targets -> existe liquidez oposta ainda não varrida, a pelo menos
                        // targetMinR de distância e sem FVG HTF contrário no caminho.
                        // Delta Imbalance -> o candle de confirmação entregou volume desequilibrado
                        // a favor da direção do setup.
                        float tgtP  = conf_FindTarget(ifvgBullish, lvl)
                        float riskT = math.abs(lvl - slPrice)
                        tr.tgtPrice := tgtP
                        tr.clearTgt := not na(tgtP) and (targetMinR <= 0 or math.abs(tgtP - lvl) >= riskT * targetMinR) and conf_PathClear(ifvgBullish, lvl, tgtP)
                        tr.deltaVal := deltaRatioPrev
                        tr.deltaImb := not na(deltaRatioPrev) and (ifvgBullish ? deltaRatioPrev >= deltaMinRatio : deltaRatioPrev <= -deltaMinRatio)

                        if showSL
                            tr.slLine  := line.new(bestBar, slPrice, bar_index + 3, slPrice, color = color.new(colorSL, 35), style = line.style_dotted)
                            tr.slLabel := label.new(bar_index + 3, slPrice, "SL", style = label.style_label_left, color = noColor, textcolor = colorSL, size = size.small, text_font_family = font.family_monospace)
                            ict_CreateRR(tr, x1, x2)

                        // Nível de validade (B.E ou 1R). É SEMPRE calculado, independente do toggle
                        // "Break Even" (showBE), pois o ict_ManageTrades depende dele para decidir
                        // quando o setup vira "válido" e, dependendo de freezeAfterValid, se ele
                        // deve ou não continuar sendo checado contra o SL depois disso.
                        float riskRangeValid = math.abs(lvl - slPrice)
                        float validPrice = switch validityCriteria
                            "B.E" => e.swingPrice
                            "1R"  => ifvgBullish ? lvl + riskRangeValid : lvl - riskRangeValid
                        tr.validPrice := validPrice

                        if showBE
                            string validLabelTxt = validityCriteria == "1R" ? "1R" : "B.E"
                            int    validLineX1   = validityCriteria == "1R" ? x1 : e.swingBar
                            tr.bePrice := validPrice
                            tr.beLine  := line.new(validLineX1, validPrice, x2, validPrice, color = color.new(colorBE, 35), style = line.style_dotted)
                            tr.beLabel := label.new(x2, validPrice, validLabelTxt, style = label.style_label_left, color = noColor, textcolor = colorBE, size = size.small, text_font_family = font.family_monospace)

                        trades.push(tr)
                        setupRecords.push(tr)
                        // Espelho de 1 posicao para a tabela - ver o comentario no [State].
                        lastSetup.clear()
                        lastSetup.push(tr)
                        while setupHistory > 0 and setupRecords.size() > setupHistory
                            Trade oldest = setupRecords.shift()
                            int oldIdx = trades.indexof(oldest)
                            if oldIdx >= 0
                                trades.remove(oldIdx)
                            box.delete(oldest.fvgBox)
                            line.delete(oldest.entryLine)
                            label.delete(oldest.entryLabel)
                            line.delete(oldest.slLine)
                            label.delete(oldest.slLabel)
                            line.delete(oldest.beLine)
                            label.delete(oldest.beLabel)
                            ict_DeleteRR(oldest)

                        if enableAlerts and inAlertSession
                            [alHits, alTot] = conf_Score(tr)
                            [alGrade, alGcol] = conf_Grade(alHits, alTot)
                            string confTxt = " | Nota " + alGrade + " (" + str.tostring(alHits) + "/" + str.tostring(alTot) + ")" + (tr.sweptHTF ? " | Sweep" : "") + (tr.smtOk ? " | SMT" : "") + (tr.htfFvgOk ? " | HTF FVG" : "") + (tr.deltaImb ? " | Delta" : "") + (tr.clearTgt ? " | Alvo" : "") + (tr.volSpike ? " | Vol" : "")
                            alert((ifvgBullish ? "Bullish" : "Bearish") + " IFVG confirmed on " + syminfo.ticker + confTxt, alert.freq_once_per_bar_close)
                if not trigger and bar_index - e.fvgBar1 > graceCandles
                    arr.remove(i)

ict_ManageTrades(array<Trade> trades) =>
    int n = trades.size()
    if n > 0
        for i = n - 1 to 0
            Trade tr = trades.get(i)
            if not tr.resolved
                bool slHit    = tr.bullish ? (not na(tr.slPrice) and low <= tr.slPrice) : (not na(tr.slPrice) and high >= tr.slPrice)
                bool validHit = tr.bullish ? (not na(tr.validPrice) and high >= tr.validPrice) : (not na(tr.validPrice) and low <= tr.validPrice)

                // Se "Congelar após atingir critério" estiver ligado (freezeAfterValid), um setup que já
                // atingiu o nível de validade (validReached) para de ser apagado por SL.
                // Se estiver desligado, o SL continua valendo mesmo depois de validReached.
                bool deleteOnSL = slHit and (not freezeAfterValid or not tr.validReached)

                if deleteOnSL
                    tr.resolved := true
                    tr.stopped  := true
                    line.delete(tr.slLine)
                    label.delete(tr.slLabel)
                    line.delete(tr.beLine)
                    label.delete(tr.beLabel)
                    ict_DeleteRR(tr)
                    box.delete(tr.fvgBox)
                    line.delete(tr.entryLine)
                    label.delete(tr.entryLabel)
                    int recIdx = setupRecords.indexof(tr)
                    if recIdx >= 0
                        setupRecords.remove(recIdx)
                    trades.remove(i)
                else if validHit and not tr.validReached
                    tr.validReached := true
                    box.set_border_color(tr.fvgBox, colorBE)
                    line.set_color(tr.entryLine, colorBE)
                    label.set_textcolor(tr.entryLabel, colorBE)
                    label.set_color(tr.entryLabel, noColor)
                    if freezeAfterValid
                        tr.resolved := true
                        trades.remove(i)
                    // Se freezeAfterValid = false, o trade continua no array `trades`
                    // (resolved = false) para seguir sendo monitorado contra o SL nos próximos bars.
//#endregion


//#region[Functions and Methods - Sweep Detector]
// Apelido de timeframe do gráfico. O 2m não tem escala própria na tabela abaixo: é
// tratado como se fosse o 3m (mesmo HTF). Sem isso ele cairia no ramo default e
// puxaria o mensal.
autoHTF() =>
    string tfBase = timeframe.period == "2" ? "3" : timeframe.period
    switch tfBase
        "1"   => "15"
        "3"   => "30"
        "5"   => "60"
        "15"  => "240"
        "60"  => "1D"
        "240" => "1W"
        "1D"  => "1M"
        =>      "1M"

method addPoint(SwingSet this, float y, int t) =>
    this.points.push(SwingPoint.new(y, t))
    if this.points.size() > MAX_SWING_HISTORY
        this.points.remove(0)

method findSweep(SwingSet this, float testPrice, bool isHigh) =>
    bool  found  = false
    float foundY = na
    int   foundT = na
    if this.points.size() > 0
        for i = this.points.size() - 1 to 0
            pt = this.points.get(i)
            bool isSwept = isHigh ? testPrice > pt.y : testPrice < pt.y
            if isSwept
                bool better = not found or (isHigh ? pt.y > foundY : pt.y < foundY)
                if better
                    foundY := pt.y
                    foundT := pt.t
                found := true
                this.points.remove(i)
    [found, foundY, foundT]
//#endregion


//#region[Functions and Methods - HTF FVG]
// Ponta esquerda de cada borda, do jeito que o FADI faz (flo/fhi): procura, entre a barra
// atual e o tempo do candle HTF de origem, a barra do GRÁFICO cujo pavio encostou exatamente
// no nível — é ela que ancora a linha. Assim cada borda começa no candle que a criou, e não
// as duas juntas na abertura do gap. `newest` escolhe a batida mais recente (borda antiga) ou
// a mais antiga (borda nova) do intervalo; sem batida exata, cai no tempo padrão.
htfFvg_Anchor(float p, int st, int dflt, bool useHigh, bool newest) =>
    int loc = dflt
    int n   = (time - st) / htfFvgMs
    if n >= 0 and n < 5000
        if newest
            for i = 0 to n
                if (useHigh ? high[i] : low[i]) == p
                    loc := time[i]
                    break
        else
            for i = n to 0
                if (useHigh ? high[i] : low[i]) == p
                    loc := time[i]
                    break
    loc

htfFvg_Del(HtfFVG f) =>
    line.delete(f.lnA)
    line.delete(f.lnB)
    line.delete(f.lnCe)
    linefill.delete(f.fill)
    label.delete(f.lbl)

// Move a ponta direita da zona. A CE para antes (htfFvgCeGap) para não cruzar o label, e nunca
// recua para trás da própria origem.
htfFvg_SetRight(HtfFVG f, int x) =>
    line.set_x2(f.lnA, x)
    line.set_x2(f.lnB, x)
    line.set_x2(f.lnCe, int(math.max(x - htfFvgCeGap, line.get_x1(f.lnCe))))
    if not na(f.lbl)
        label.set_x(f.lbl, x)

// As linhas de borda continuam invisíveis (transparência 100) — só o que é visível troca de cor.
htfFvg_SetColor(HtfFVG f, color c) =>
    line.set_color(f.lnA, color.new(c, 100))
    line.set_color(f.lnB, color.new(c, 100))
    line.set_color(f.lnCe, color.new(c, 60))
    if not na(f.fill)
        linefill.set_color(f.fill, color.new(c, HTF_FVG_BG))
    if not na(f.lbl)
        label.set_textcolor(f.lbl, c)

// Desenho da zona: geometria do FADI (cada borda ancorada na barra que a criou, projeção à
// direita, CE pontilhada saindo do meio do intervalo das âncoras), mas SEM contorno — as duas
// linhas de borda são 100% transparentes e servem só de trilho para o linefill, e a tampa
// vertical da direita não existe. O label fica POR DENTRO da ponta direita: `style_label_right`
// põe o corpo do label à esquerda da âncora (o `style_label_left` jogaria para fora da zona).
// A CE termina htfFvgCeGap antes dessa ponta, senão passaria por baixo do texto do label.
htfFvg_Add(array<HtfFVG> arr, bool bullish, int tOld, int tNew, float zTop, float zBot, string tfTxt) =>
    color base = bullish ? htfFvgBullColor : htfFvgBearColor
    float pA   = bullish ? zBot : zTop
    float pB   = bullish ? zTop : zBot
    float mid  = (zTop + zBot) / 2
    int   xA   = htfFvg_Anchor(pA, tOld, tOld, bullish, true)
    int   xB   = htfFvg_Anchor(pB, tOld, tNew, not bullish, false)
    int   xR   = time + htfFvgMs * (htfFvgExt + 1)
    color inv  = color.new(base, 100)
    line  lnA  = line.new(xA, pA, xR, pA, xloc = xloc.bar_time, style = line.style_solid, color = inv, width = 1)
    line  lnB  = line.new(xB, pB, xR, pB, xloc = xloc.bar_time, style = line.style_solid, color = inv, width = 1)
    linefill fl = linefill.new(lnA, lnB, color.new(base, HTF_FVG_BG))
    int   xCe1 = (xA + xB) / 2
    line  lnCe = line.new(xCe1, mid, int(math.max(xR - htfFvgCeGap, xCe1)), mid, xloc = xloc.bar_time, style = line.style_dotted, color = color.new(base, 60), width = 1)
    label lb   = showHtfFvgLabel ? label.new(xR, mid, tfTxt, xloc = xloc.bar_time, style = label.style_label_right, color = noColor, textcolor = base, size = size.small, text_font_family = font.family_monospace) : na
    arr.push(HtfFVG.new(lnA, lnB, lnCe, fl, lb, zTop, zBot, mid, bullish))
    while arr.size() > htfFvgCount
        HtfFVG old = arr.shift()
        htfFvg_Del(old)

// Estende, aplica a invalidação e devolve `true` se o candle atual tocou algum FVG ainda aberto.
// A invalidação segue o padrão FADI: avaliada SÓ no fechamento de um candle do timeframe maior
// (checkMit) e SÓ pelo corpo desse candle (bodyLo/bodyHi = min/max de open e close). O pavio
// não invalida nada, e o corpo precisa fechar além da borda de ORIGEM do gap — ou seja,
// atravessar a zona por inteiro, não apenas encostar no lado mais próximo.
// Só a geometria da ponta direita espera a última barra (como o UPZ do FADI): no histórico ela
// seria sobrescrita de qualquer jeito. O toque e a invalidação continuam sendo avaliados barra
// a barra, porque alimentam a tabela de confluência.
htfFvg_Manage(array<HtfFVG> arr, bool checkMit, float bodyLo, float bodyHi) =>
    bool touched = false
    int  xR      = time + htfFvgMs * (htfFvgExt + 1)
    if arr.size() > 0
        for i = arr.size() - 1 to 0
            HtfFVG f = arr.get(i)
            if not f.filled
                if low <= f.top and high >= f.bottom
                    touched := true
                bool full = checkMit and (f.bullish ? bodyLo < f.bottom : bodyHi > f.top)
                if full
                    if htfFvgKeepFilled
                        // Para de se estender e fica cinza na barra em que foi invalidado.
                        f.filled := true
                        htfFvg_SetRight(f, time)
                        htfFvg_SetColor(f, htfFvgFilledColor)
                    else
                        htfFvg_Del(f)
                        arr.remove(i)
                else if barstate.islast
                    htfFvg_SetRight(f, xR)
    touched
//#endregion

smt_fss(x) =>

// ---------------------------------------------------------------------------------------
// SMT de pavios adjacentes (motor do PO3)
// ---------------------------------------------------------------------------------------
// Um sinal de pavio vale enquanto o ativo correlacionado NÃO tomar o próprio nível. Quando
// ele finalmente toma, a divergência deixou de existir e o desenho sai do gráfico.
smt_finv(line[] la, label[] lba, float[] ra, bool isUp, float cv) =>
    i = array.size(la) - 1
    while i >= 0
        if isUp ? cv > array.get(ra, i) : cv < array.get(ra, i)
            ln = array.get(la, i)
            lb = array.get(lba, i)
            if not na(ln)
                line.delete(ln)
            if not na(lb)
                label.delete(lb)
            array.remove(la, i)
            array.remove(lba, i)
            array.remove(ra, i)
        i := i - 1

// ---------------------------------------------------------------------------------------
// SMT de pivô (motor do PO3)
// ---------------------------------------------------------------------------------------
// Compara o pivô atual com o anterior nas duas séries. Cada chamada carrega o próprio
// estado de série (o Pine faz inline), então as três escalas não se misturam.
smt_gd(ph, y2, sy2, ln, col, lbl, float[] dedup) =>
    var float y1 = na
    var float sy1 = na
    var int x1 = na
    var float xt = na
    var smt = 0
    if y2 != y2[1] and sy2 != sy2[1]
        if (y2 - y1) * (sy2 - sy1) < 0
            dup = array.includes(dedup, y2)
            if not dup
                // Sinais ja sao opostos (teste acima), entao o lado do grafico decide o grau:
                // num topo, y2 > y1 = o grafico TOMOU o pivo anterior e o par nao acompanhou.
                bool  _strong = ph == (y2 > y1)
                color _sCol   = _strong ? col : color.new(col, smt_wkFade)
                if _strong or showSmtWeak
                    array.push(smt_pvLn, line.new(smt_n[ln], y2, x1, y1, color=_sCol, width=smt_lw, style=smt_lsL))
                    array.push(smt_pvLb, smt_showLabels ? label.new(int(math.round((xt + time[ln]) / 2.0)), (y1 + y2) / 2.0, lbl, xloc=xloc.bar_time, style=ph ? label.style_label_down : label.style_label_up, color=smt_W100, textcolor=_sCol, size=smt_lzL) : na)
                    while array.size(smt_pvLn) > smt_mxShow
                        line.delete(array.shift(smt_pvLn))
                        label.delete(array.shift(smt_pvLb))
                array.push(dedup, y2)
                // Só o grau FORTE conta como confluência na tabela.
                if _strong
                    array.set(smtRefBars, ph ? 1 : 0, smt_n[ln])
                if array.size(dedup) > 20
                    array.shift(dedup)
                smt += 1
        sy1 := sy2
        y1 := y2
        x1 := smt_n[ln]
        xt := time[ln]
    else if (ph and y2 > y2[1]) or (not ph and y2 < y2[1])
        sy1 := na
        y1 := y2
        x1 := smt_n[ln]
        xt := time[ln]
    smt

// ---------------------------------------------------------------------------------------
// SMT de FVG (motor do PO3)
// ---------------------------------------------------------------------------------------
// Acompanha o FVG mais recente de um lado numa série e avisa quando ele acabou de ser
// mitigado.
smt_trackFVG(SmtFVG st, h, l, isBull) =>
    justMit = false
    formed = isBull ? (l > h[2]) : (h < l[2])
    if formed
        st.top := isBull ? l : l[2]
        st.bot := isBull ? h[2] : h
        st.fbar := bar_index[2]
        st.mit := false
    else if not st.mit
        if isBull ? (l <= st.top) : (h >= st.bot)
            st.mit := true
            justMit := true
    justMit

smt_fBox(SmtFVG st, isBull) =>
    if smt_fShowBox
        if not st.mit and not na(st.top)
            if na(st.bx)
                bcol = isBull ? color.new(#75887F, smt_fBoxTr) : color.new(#967575, smt_fBoxTr)
                st.bx := box.new(st.fbar, st.top, bar_index, st.bot, border_color=na, bgcolor=bcol)
            else
                box.set_right(st.bx, bar_index)
        else if st.mit and not na(st.bx)
            box.delete(st.bx)
            st.bx := na
    true

// Um ativo mitigou o FVG dele e o outro continua com o FVG aberto = divergência.
smt_chkFVGSMT(mainJust, SmtFVG other, SmtFVG my, touchY, isBull, col, lbl) =>
    sig = false
    if mainJust and not other.mit and (bar_index - other.fbar) <= smt_fMaxAge
        if not na(my.top) and not na(my.bot)
            box _fsb = na
            if smt_fShowBox
                _fsb := box.new(my.fbar, my.top, bar_index, my.bot, border_color=na, bgcolor=color.new(col, smt_fBoxTr))
            leftY = isBull ? my.top : my.bot
            leftX = my.fbar + 2
            _fsl = line.new(leftX, leftY, bar_index, touchY, color=col, width=smt_lw, style=line.style_dotted)
            label _fslb = na
            if smt_showLabels
                _fvgOff = bar_index - leftX
                _fvgLblX = int(math.round((time[_fvgOff] + time) / 2.0))
                _fslb := label.new(_fvgLblX, (leftY + touchY) / 2.0, lbl, xloc=xloc.bar_time, style=isBull ? label.style_label_up : label.style_label_down, color=smt_W100, textcolor=col, size=smt_lzL)
            array.push(smt_fsBx, _fsb), array.push(smt_fsLn, _fsl), array.push(smt_fsLb, _fslb)
            while array.size(smt_fsLn) > smt_fMxShow
                box.delete(array.shift(smt_fsBx))
                line.delete(array.shift(smt_fsLn))
                label.delete(array.shift(smt_fsLb))
            sig := true
    sig

var SmtArrHD smt_a1 = SmtArrHD.new(array.new_line(), array.new_label(), array.new_float(), array.new_line(), array.new_label(), array.new_float())

smt_c2lp = smt_inv2 ? smt_c2h[1] : smt_c2l[1]

smt_c2ln = smt_inv2 ? smt_c2h : smt_c2l

smt_fpa(smt_eA and smt_e2, smt_ud2, smt_ud2[1], smt_dd2, smt_dd2[1], high, low, high[1], low[1], smt_c2hp, smt_c2lp, smt_c2hn, smt_c2ln, smt_a2, smt_t2, smt_c2)

    if smt_e2

        smt_gdSet(rph, rpl, rphL, rplL, rphS, rplS, smt_inv2 ? c2pl : c2ph, smt_inv2 ? c2ph : c2pl, smt_inv2 ? c2plL : c2phL, smt_inv2 ? c2phL : c2plL, smt_inv2 ? c2plS : c2phS, smt_inv2 ? c2phS : c2plS, smt_c2, smt_t2, smt_dedH2, smt_dedL2)
