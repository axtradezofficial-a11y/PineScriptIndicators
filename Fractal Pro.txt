detectCISDAndProjections(int c2_bar, float c2_extreme_level, bool is_bullish, int end_bar) =>
    series_high = usebody_for_confirmation ? math.max(open[bar_index - c2_bar], close[bar_index - c2_bar]) : c2_extreme_level
    series_low = usebody_for_confirmation ? math.min(open[bar_index - c2_bar], close[bar_index - c2_bar]) : c2_extreme_level
    series_count = 0

    series_count := 1

    float true_extreme_val = is_bullish ? low[bar_index - c2_bar] : high[bar_index - c2_bar]
    int true_extreme_bar = c2_bar

    for i = 1 to 20
        check_bar = c2_bar - i
        if check_bar < 0
            break

        is_bullish_candle = close[bar_index - check_bar] > open[bar_index - check_bar]

        if is_bullish
            if not is_bullish_candle
                candle_high = usebody_for_confirmation ? math.max(open[bar_index - check_bar], close[bar_index - check_bar]) : high[bar_index - check_bar]
                candle_low = usebody_for_confirmation ? math.min(open[bar_index - check_bar], close[bar_index - check_bar]) : low[bar_index - check_bar]
                series_high := math.max(series_high, candle_high)
                series_low := math.min(series_low, candle_low)
                if low[bar_index - check_bar] < true_extreme_val
                    true_extreme_val := low[bar_index - check_bar]
                    true_extreme_bar := check_bar
                series_count += 1
            else
                break
        else
            if is_bullish_candle
                candle_high = usebody_for_confirmation ? math.max(open[bar_index - check_bar], close[bar_index - check_bar]) : high[bar_index - check_bar]
                candle_low = usebody_for_confirmation ? math.min(open[bar_index - check_bar], close[bar_index - check_bar]) : low[bar_index - check_bar]
                series_high := math.max(series_high, candle_high)
                series_low := math.min(series_low, candle_low)
                if high[bar_index - check_bar] > true_extreme_val
                    true_extreme_val := high[bar_index - check_bar]
                    true_extreme_bar := check_bar
                series_count += 1
            else
                break

    series_broken = false
    break_bar = 0

    for i = 0 to (bar_index - c2_bar)
        check_bar = c2_bar + i
        if check_bar > bar_index
            break

        if is_bullish
            if close[bar_index - check_bar] > series_high
                series_broken := true
                break_bar := check_bar
                break
        else
            if close[bar_index - check_bar] < series_low
                series_broken := true
                break_bar := check_bar
                break

    if series_broken
        bool dir_cisd_enabled = is_bullish ? ui_bull_cisd_en : ui_bear_cisd_en
        if dir_cisd_enabled
            color  dir_cisd_color = is_bullish ? ui_bull_cisd_col : ui_bear_cisd_col
            int    dir_cisd_width = is_bullish ? ui_bull_cisd_wid : ui_bear_cisd_wid
            string dir_cisd_sty   = is_bullish ? ui_bull_cisd_sty : ui_bear_cisd_sty
            string dir_cisd_style = dir_cisd_sty == "Dashed" ? line.style_dashed : dir_cisd_sty == "Dotted" ? line.style_dotted : line.style_solid

            series_extreme = is_bullish ? series_high : series_low
            last_series_bar = c2_bar - (series_count - 1)
            cisd_line = line.new(last_series_bar, series_extreme, break_bar, series_extreme, color=dir_cisd_color, style=dir_cisd_style, width=dir_cisd_width)
            array.unshift(cisd_lines, cisd_line)

            if array.size(cisd_lines) > ui_cisd_max
                line.delete(array.pop(cisd_lines))

            if ui_prot_swing_en
                string label_sz_input = ui_prot_swing_sz == "Tiny" ? size.tiny : ui_prot_swing_sz == "Small" ? size.small : ui_prot_swing_sz == "Normal" ? size.normal : ui_prot_swing_sz == "Large" ? size.large : size.huge
                string lbl_style = is_bullish ? label.style_label_up : label.style_label_down
                prot_swing_lbl = label.new(true_extreme_bar, true_extreme_val, "●", color=color.new(color.white, 100), textcolor=dir_cisd_color, style=lbl_style, size=label_sz_input)
                array.unshift(protected_swing_labels, prot_swing_lbl)
                
                if array.size(protected_swing_labels) > ui_cisd_max
                    label.delete(array.pop(protected_swing_labels))

        if show_projections
            if array.size(projection_lines) > 0
                for ln in projection_lines
                    line.delete(ln)
                array.clear(projection_lines)
            if array.size(projection_labels) > 0
                for lbl in projection_labels
                    label.delete(lbl)
                array.clear(projection_labels)

            series_range = series_high - series_low
            if series_range > 0
                break_price = is_bullish ? series_high : series_low
                left_anchor_bar = c2_bar - (series_count - 1)
                levels = str.split(projection_levels, ",")
                proj_line_style = ui_proj_ln_sty == "Dashed" ? line.style_dashed : ui_proj_ln_sty == "Solid" ? line.style_solid : line.style_dotted
                for i = 0 to array.size(levels) - 1
                    level_str = str.trim(array.get(levels, i))
                    if level_str != ""
                        level = str.tonumber(level_str)
                        if not na(level)
                            proj_price = is_bullish ? break_price + (series_range * level) : break_price - (series_range * level)
                            if ui_proj_ln_en
                                proj_line = line.new(left_anchor_bar, proj_price, bar_index, proj_price, color=projection_color, style=proj_line_style, width=ui_proj_ln_wid)
                                array.unshift(projection_lines, proj_line)
                            if ui_proj_lbl_en
                                string formatted_level = level > 0 ? "-" + str.tostring(level) : str.tostring(level)
                                proj_label = label.new(left_anchor_bar, proj_price, formatted_level, color=color.new(color.white, 100), textcolor=projection_color, style=label.style_label_right, size=projection_label_size, text_font_family=font.family_monospace)
                                array.unshift(projection_labels, proj_label)

method ValidTimeframe(Helper helper, string HTF) =>
    helper.name := HTF
    if timeframe.in_seconds(HTF) >= timeframe.in_seconds("1D") and timeframe.in_seconds(HTF) > timeframe.in_seconds()
        true
    else
        n1 = timeframe.in_seconds()
        n2 = timeframe.in_seconds(HTF)
        n3 = n1 % n2
        (n1 < n2 and math.round(n2/n1) == n2/n1)

method RemainingTime(Helper helper, string HTF) =>
    helper.name     := HTF
    if barstate.isrealtime
        timeRemaining   = (time_close(HTF) - timenow)/1000
        days            = math.floor(timeRemaining / 86400)
        hours           = math.floor((timeRemaining - (days*86400)) / 3600)
        minutes         = math.floor((timeRemaining - (days*86400) - (hours*3600))/ 60)
        seconds         = math.floor(timeRemaining - (days*86400) - (hours*3600) - (minutes*60))

        r = str.tostring(seconds, "00")
        if minutes > 0 or hours > 0 or days > 0
            r := str.tostring(minutes, "00") + ":" + r
        if hours > 0 or days > 0
            r := str.tostring(hours, "00") + ":" + r
        if days > 0
            r := str.tostring(days) + "D " + r
        r
    else
        "n/a"

method HTFName(Helper helper, string HTF) =>
    helper.name := "HTFName"
    formatted = HTF
    seconds = timeframe.in_seconds(HTF)
    if seconds < 60
        formatted := str.tostring(seconds) + "S"
    else if (seconds / 60) < 60
        formatted := str.tostring((seconds/60)) + "M"
    else if (seconds/60/60) < 24
        formatted := str.tostring((seconds/60/60)) + "H"
    formatted

    else if (current_seconds/60/60) < 24

        str.tostring((current_seconds/60/60/24)) + "D"

method GetHTFBias(Helper helper, Candle last_closed, Candle prev_closed) =>
    helper.name := "GetHTFBias"

    if na(last_closed) or na(prev_closed)
        "Bias: Neutral"
    else
        last_close = last_closed.c
        prev_close = prev_closed.c
        last_high = last_closed.h
        last_low = last_closed.l
        prev_high = prev_closed.h
        prev_low = prev_closed.l
        last_open = last_closed.o

        swept_high = last_high > prev_high
        swept_low = last_low < prev_low

        if last_close > prev_high
            "Bias: Bullish"
        else if last_close < prev_low
            "Bias: Bearish"
        else if swept_high and last_close < prev_high
            "Bias: Bearish"
        else if swept_low and last_close > prev_low
            "Bias: Bullish"
        else if not swept_high and not swept_low
            "Bias: Neutral"
        else if swept_high and swept_low
            upper_wick = last_high - math.max(last_open, last_close)
            lower_wick = math.min(last_open, last_close) - last_low
            if lower_wick > upper_wick
                "Bias: Bearish"
            else if upper_wick > lower_wick
                "Bias: Bullish"
            else
                "Bias: Neutral"
        else
            "Bias: Neutral"

method calculateLogMidpoint(float high, float low, float open, float close) =>
    float log_high = math.log(high)
    float log_low = math.log(low)
    float log_open = math.log(open)
    float log_close = math.log(close)

    float body_size = math.abs(log_close - log_open)
    float upper_wick = log_high - math.max(log_open, log_close)
    float lower_wick = math.min(log_open, log_close) - log_low

    float log_mid_level = if math.max(upper_wick, lower_wick) > body_size
        if upper_wick > lower_wick
            log_high - upper_wick/2
        else
            log_low + lower_wick/2
    else
        (log_high + log_low)/2

    math.exp(log_mid_level)

method FindImbalance(CandleSet candleSet, bool isPrimary, bool drawSweep) =>
    if barstate.isrealtime or barstate.islast
        while candleSet.imbalances.size() > 0
            Imbalance del = candleSet.imbalances.pop()
            box.delete(del.b)

        if candleSet.candles.size() > 3 and settings.fvg_show
            if isPrimary
                for i = 0 to candleSet.candles.size() - 2
                    current_candle = candleSet.candles.get(i)
                    prev_candle = candleSet.candles.get(i + 1)
                    float mid_level = calculateLogMidpoint(prev_candle.h, prev_candle.l, prev_candle.o, prev_candle.c)
                    midpoint_line = line.new(box.get_left(prev_candle.body) + settings.width/2, mid_level,box.get_left(current_candle.body) + settings.width/2, mid_level,color=TEMPLATE_COLOR, width=1, style=line.style_dotted)
                    midpoint_lines.unshift(midpoint_line)

            if drawSweep
                for i = 0 to candleSet.candles.size() - 2
                    current_candle = candleSet.candles.get(i)
                    next_candle = candleSet.candles.get(i + 1)
                    if current_candle.h > next_candle.h and current_candle.c < next_candle.h
                        sweep_line = line.new(box.get_left(next_candle.body) + settings.width/2, next_candle.h, box.get_right(current_candle.body) + settings.width, next_candle.h, color=TEMPLATE_COLOR, width=1)
                        sweep_lines.unshift(sweep_line)
                    if current_candle.l < next_candle.l and current_candle.c > next_candle.l
                        sweep_line = line.new(box.get_left(next_candle.body) + settings.width/2, next_candle.l, box.get_right(current_candle.body) + settings.width, next_candle.l, color=TEMPLATE_COLOR, width=1)
                        sweep_lines.unshift(sweep_line)

        if candleSet.candles.size() > 3 and settings.fvg_show
            for i = 0 to candleSet.candles.size() -3
                candle1 = candleSet.candles.get(i)
                candle2 = candleSet.candles.get(i+2)
                candle3 = candleSet.candles.get(i+1)

method Monitor(CandleSet candleSet, bool drawLines) =>
    HTFBarTime = time(candleSet.settings.htf)
    isNewHTFCandle = use_actual_day_change and candleSet.settings.htf == "1D" ? dayofmonth(time) != dayofmonth(time[1]) : ta.change(HTFBarTime) != 0

    if isNewHTFCandle
        if drawLines and settings.htf_line_show
            float visual_offset = label_atr * htf_line_extension
            htf_lines.unshift(line.new(bar_index, high + visual_offset, bar_index, low - visual_offset, color=settings.htf_line_color, style=settings.htf_line_style, width=settings.htf_line_width, extend=extend.none))

            float htf_sec = timeframe.in_seconds(candleSet.settings.htf)
            string timeStr = ""

            if htf_sec >= 2419200
                string m_name = switch month(time)
                    1 => "Jan"
                    2 => "Feb"
                    3 => "Mar"
                    4 => "Apr"
                    5 => "May"
                    6 => "Jun"
                    7 => "Jul"
                    8 => "Aug"
                    9 => "Sep"
                    10 => "Oct"
                    11 => "Nov"
                    12 => "Dec"
                    => ""
                timeStr := m_name
            else if htf_sec >= 604800
                string m_name = switch month(time)
                    1 => "Jan"
                    2 => "Feb"
                    3 => "Mar"
                    4 => "Apr"
                    5 => "May"
                    6 => "Jun"
                    7 => "Jul"
                    8 => "Aug"
                    9 => "Sep"
                    10 => "Oct"
                    11 => "Nov"
                    12 => "Dec"
                    => ""
                timeStr := str.tostring(dayofmonth(time)) + " " + m_name
            else if htf_sec >= 86400
                string d_name = switch dayofweek(time)
                    1 => "Sun"
                    2 => "Mon"
                    3 => "Tue"
                    4 => "Wed"
                    5 => "Thu"
                    6 => "Fri"
                    7 => "Sat"
                    => ""
                timeStr := d_name
            else
                timeStr := str.tostring(hour(time), "00") + ":" + str.tostring(minute(time), "00")

            htf_labels.unshift(label.new(bar_index, high + visual_offset, text=timeStr, color=color.new(color.white, 100), textcolor=settings.htf_line_color, style=label.style_label_down, size=size.tiny))

        if htf_lines.size() > candleSet.settings.max_display
            line.delete(htf_lines.pop())
        if htf_labels.size() > candleSet.settings.max_display
            label.delete(htf_labels.pop())
        if htf_labels_bottom.size() > candleSet.settings.max_display
            label.delete(htf_labels_bottom.pop())

        Candle candle = Candle.new()
        candle.o := open, candle.c := close, candle.h := high, candle.l := low
        candle.o_idx := bar_index, candle.c_idx := bar_index, candle.h_idx := bar_index, candle.l_idx := bar_index
