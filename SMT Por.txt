f_is_mid_prefix(string px) =>
    px == "CAPITALCOM" or px == "CMCMARKETS"

f_is_chicago_prefix(string px) =>
    px == "CME" or px == "CBOT" or px == "CBOT_MINI" or px == "CBOT_DL" or px == "CME_MINI" or px == "CME_DL" or px == "CME_MINI_DL" or px == "COMEX" or px == "NYMEX"

f_is_utc_prefix(string px) =>
    px == "BINANCE" or px == "BYBIT" or px == "KRAKEN" or px == "GATE.IO" or px == "BITSTAMP" or px == "COINBASE" or px == "GEMINI"

f_oracle_tz(string px) =>
    f_is_chicago_prefix(px) ? "America/Chicago" : f_is_mid_prefix(px) ? "Etc/GMT+5" : f_is_utc_prefix(px) ? "UTC" : "America/New_York"

f_oracle_day_start_hour(string px) =>
    f_is_chicago_prefix(px) or f_is_mid_prefix(px) ? 17 : f_is_utc_prefix(px) ? 22 : 18

f_days_since_sunday(int dow) =>
    dow == dayofweek.sunday ? 0 : dow == dayofweek.monday ? 1 : dow == dayofweek.tuesday ? 2 : dow == dayofweek.wednesday ? 3 : dow == dayofweek.thursday ? 4 : dow == dayofweek.friday ? 5 : 6

f_smt_anchor_range_key(int barTime, int anchorTime, int rangeMinutes) =>
    int rangeMs = rangeMinutes * 60 * 1000
    int bucket = int(math.floor((barTime - anchorTime) / rangeMs))
    anchorTime + bucket * rangeMs

f_smt_daily_range_key(int dayStartTime) =>
    dayStartTime

f_smt_iso_week_year(int refTime, string tz) =>
    int wy = year(refTime, tz)
    int wm = month(refTime, tz)
    int cw = weekofyear(refTime, tz)
    if wm == 12 and cw == 1
        wy := wy + 1
    else if wm == 1 and cw >= 52
        wy := wy - 1
    wy

f_smt_iso_week_range_key(int dayStartTime, int oneDayMs, string tz) =>
    int refTime = dayStartTime + oneDayMs
    int wy = f_smt_iso_week_year(refTime, tz)
    int cw = weekofyear(refTime, tz)
    wy * 100 + cw

f_smt_month_range_key(int dayStartTime, string tz) =>
    year(dayStartTime, tz) * 100 + month(dayStartTime, tz)

f_smt_month_of_quarter_slot(int dayStartTime, string tz) =>
    int cm = month(dayStartTime, tz)
    int qsM = int(math.floor((cm - 1) / 3)) * 3 + 1
    cm - qsM

f_smt_month_of_quarter_range_key(int dayStartTime, string tz) =>
    int cy = year(dayStartTime, tz)
    int cm = month(dayStartTime, tz)
    int q = int(math.floor((cm - 1) / 3)) + 1
    int slot = f_smt_month_of_quarter_slot(dayStartTime, tz)
    cy * 100 + q * 10 + slot

f_smt_quarter_subrange_slot(int barTime, int dayStartTime, string tz, int startHour) =>
    int cy = year(dayStartTime, tz)
    int cm = month(dayStartTime, tz)
    int qsM = int(math.floor((cm - 1) / 3)) * 3 + 1
    int qsT = timestamp(tz, cy, qsM, 1, startHour, 0)
    int qeT = qsM == 10 ? timestamp(tz, cy + 1, 1, 1, startHour, 0) : timestamp(tz, cy, qsM + 3, 1, startHour, 0)
    int q = int(math.floor((barTime - qsT) * 4.0 / math.max(1, qeT - qsT)))
    math.max(0, math.min(3, q))

f_smt_quarter_subrange_key(int barTime, int dayStartTime, string tz, int startHour) =>
    int cy = year(dayStartTime, tz)
    int cm = month(dayStartTime, tz)
    int quarter = int(math.floor((cm - 1) / 3)) + 1
    int slot = f_smt_quarter_subrange_slot(barTime, dayStartTime, tz, startHour)
    cy * 100 + quarter * 10 + slot

f_smt_four_year_phase(int y) =>
    y % 4 == 3 ? 0 : y % 4 == 0 ? 1 : y % 4 == 1 ? 2 : 3

f_smt_four_year_range_key(int barTime, string tz) =>
    int y = year(barTime, tz)
    int block = int(math.floor(y / 4))
    block * 10 + f_smt_four_year_phase(y)

f_smt_local_day_start(int barTime, string tz, int startHour) =>
    int localYear = year(barTime, tz)
    int localMonth = month(barTime, tz)
    int localDay = dayofmonth(barTime, tz)
    int todayStart = timestamp(tz, localYear, localMonth, localDay, startHour, 0)
    int previousDayProbe = barTime - 24 * 60 * 60 * 1000
    barTime >= todayStart ? todayStart : timestamp(tz, year(previousDayProbe, tz), month(previousDayProbe, tz), dayofmonth(previousDayProbe, tz), startHour, 0)

f_smt_shift_local_day_start(int sourceDayStart, int daysBack, string tz, int startHour) =>
    int shiftedProbe = sourceDayStart - daysBack * 24 * 60 * 60 * 1000
    timestamp(tz, year(shiftedProbe, tz), month(shiftedProbe, tz), dayofmonth(shiftedProbe, tz), startHour, 0)

f_smt_body_intersects(int bx1, int bx2, float y1, float y2) =>
    bool hit = false
    int span = bx2 - bx1
    if span > 3
        int fromX = math.max(bx1 + 2, bar_index - 4999)
        int toX = math.min(bx2 - 2, bar_index)
        if fromX <= toX
            for x = fromX to toX
                int barsAgo = bar_index - x
                float linePrice = y1 + (y2 - y1) * (x - bx1) / span
                float bodyLow = math.min(open[barsAgo], close[barsAgo])
                float bodyHigh = math.max(open[barsAgo], close[barsAgo])
                if bodyLow <= linePrice and linePrice <= bodyHigh
                    hit := true
                    break
    hit

// Equality is not a liquidity take; strict extrema retain first-touch identity.
f_smt_high_swept(float currentValue, float previousValue) =>
    not na(currentValue) and not na(previousValue) and currentValue > previousValue

f_smt_low_swept(float currentValue, float previousValue) =>
    not na(currentValue) and not na(previousValue) and currentValue < previousValue

f_smt_update_state(int cycleIdx, int srcIdx, bool engineEnabled, int rangeKey, float srcHigh, float srcLow, int srcTime) =>
    if engineEnabled and not na(rangeKey)
        int idx = f_smt_state_idx(cycleIdx, srcIdx)
        SmtRangeState st = array.get(smt_range_states, idx)
        bool dataReady = not na(srcHigh) and not na(srcLow) and not na(srcTime)
        if na(st.cycleKey)
            st.cycleKey := rangeKey
            st.previousCycleKey := na
            st.hasCurrentRange := false
            st.hasPreviousRange := false
        else if rangeKey != st.cycleKey
            st.previousHigh := st.currentHigh
            st.previousLow := st.currentLow
            st.previousHighTime := st.currentHighTime
            st.previousLowTime := st.currentLowTime
            st.previousHighBar := st.currentHighBar
            st.previousLowBar := st.currentLowBar
            st.previousCycleKey := st.cycleKey
            st.hasPreviousRange := st.hasCurrentRange
            st.currentHigh := na
            st.currentLow := na
            st.currentHighTime := na
            st.currentLowTime := na
            st.currentHighBar := na
            st.currentLowBar := na
            st.hasCurrentRange := false
            st.cycleKey := rangeKey
        if dataReady and st.cycleKey == rangeKey
            if na(st.currentHigh) or srcHigh > st.currentHigh
                st.currentHigh := srcHigh
                st.currentHighTime := srcTime
                st.currentHighBar := bar_index
            if na(st.currentLow) or srcLow < st.currentLow
                st.currentLow := srcLow
                st.currentLowTime := srcTime
                st.currentLowBar := bar_index
            st.hasCurrentRange := true
        array.set(smt_range_states, idx, st)
    0

f_smt_get_state(int cycleIdx, int srcIdx) =>
    array.get(smt_range_states, f_smt_state_idx(cycleIdx, srcIdx))

// Inverse pairs are normalized once into main-domain high/low liquidity actions.
f_smt_build_sweeps(SmtRangeState st, bool inverse) =>
    bool ready = st.hasCurrentRange and st.hasPreviousRange
    float highA = inverse ? st.previousLow : st.previousHigh
    float highB = inverse ? st.currentLow : st.currentHigh
    float lowA = inverse ? st.previousHigh : st.previousLow
    float lowB = inverse ? st.currentHigh : st.currentLow
    int highTimeA = inverse ? st.previousLowTime : st.previousHighTime
    int highTimeB = inverse ? st.currentLowTime : st.currentHighTime
    int lowTimeA = inverse ? st.previousHighTime : st.previousLowTime
    int lowTimeB = inverse ? st.currentHighTime : st.currentLowTime
    int highBarA = inverse ? st.previousLowBar : st.previousHighBar
    int highBarB = inverse ? st.currentLowBar : st.currentHighBar
    int lowBarA = inverse ? st.previousHighBar : st.previousLowBar
    int lowBarB = inverse ? st.currentHighBar : st.currentLowBar
    bool highSweep = ready and (inverse ? f_smt_low_swept(highB, highA) : f_smt_high_swept(highB, highA))
    bool lowSweep = ready and (inverse ? f_smt_high_swept(lowB, lowA) : f_smt_low_swept(lowB, lowA))
    SmtSweepState.new(ready, highSweep, lowSweep, highA, highB, lowA, lowB, highTimeA, highTimeB, lowTimeA, lowTimeB, highBarA, highBarB, lowBarA, lowBarB)

f_smt_xor(bool first, bool second) =>
    (first and not second) or (not first and second)

// Pure divergence truth: exactly one synchronized market sweeps corresponding liquidity.
f_smt_resolve_pair_state(SmtSweepState mainSweep, SmtSweepState pairSweep) =>
    bool ready = mainSweep.ready and pairSweep.ready
    bool bearValid = ready and f_smt_xor(mainSweep.highSwept, pairSweep.highSwept)
    bool bullValid = ready and f_smt_xor(mainSweep.lowSwept, pairSweep.lowSwept)
    [bearValid, bullValid]

f_smt_time_to_minutes(string t) =>
    array<string> parts = str.split(t, ":")
    int h = array.size(parts) > 0 ? int(str.tonumber(array.get(parts, 0))) : 0
    int m = array.size(parts) > 1 ? int(str.tonumber(array.get(parts, 1))) : 0
    h * 60 + m

f_smt_minutes_in_window(int tMin, int startMin, int endMin) =>
    startMin <= endMin ? (tMin >= startMin and tMin <= endMin) : (tMin >= startMin or tMin <= endMin)

f_smt_session_render_ok(int pivotBTime) =>
    bool ok = true
    if smt_session_filter_effective
        ok := false
        if not na(pivotBTime)
            int pivotMinutes = hour(pivotBTime, "America/New_York") * 60 + minute(pivotBTime, "America/New_York")
            if smt_auto_session_filter
                ok := smt_auto_session_1_active ? f_smt_minutes_in_window(pivotMinutes, smt_session_1_start_min, smt_session_1_end_min) : smt_auto_session_2_active ? f_smt_minutes_in_window(pivotMinutes, smt_session_2_start_min, smt_session_2_end_min) : smt_auto_session_3_active ? f_smt_minutes_in_window(pivotMinutes, smt_session_3_start_min, smt_session_3_end_min) : smt_auto_session_4_active ? f_smt_minutes_in_window(pivotMinutes, smt_session_4_start_min, smt_session_4_end_min) : false
            else
                bool inSession1 = smt_session_1_on and f_smt_minutes_in_window(pivotMinutes, smt_session_1_start_min, smt_session_1_end_min)
                bool inSession2 = smt_session_2_on and f_smt_minutes_in_window(pivotMinutes, smt_session_2_start_min, smt_session_2_end_min)
                bool inSession3 = smt_session_3_on and f_smt_minutes_in_window(pivotMinutes, smt_session_3_start_min, smt_session_3_end_min)
                bool inSession4 = smt_session_4_on and f_smt_minutes_in_window(pivotMinutes, smt_session_4_start_min, smt_session_4_end_min)
                ok := inSession1 or inSession2 or inSession3 or inSession4
    ok

f_smt_find_event_by_id(int eventId) =>
    int idx = -1
    int i = 0
    while i < array.size(smt_events) and idx < 0
        if array.get(smt_events, i).id == eventId
            idx := i
        i += 1
    idx

f_smt_drop_render_at(int idx) =>
    if idx >= 0 and idx < array.size(smt_events)
        SmtEvent event = array.get(smt_events, idx)
        if not na(event.labelId)
            label.delete(event.labelId)
            event.labelId := na
        if not na(event.lineId)
            line.delete(event.lineId)
            event.lineId := na
        array.set(smt_events, idx, event)
    0

f_smt_delete_event_at(int idx) =>
    bool deleted = false
    if idx >= 0 and idx < array.size(smt_events)
        f_smt_drop_render_at(idx)
        array.remove(smt_events, idx)
        deleted := true
    deleted

f_smt_release_live_slot(int slot) =>
    if slot >= 0 and slot < array.size(smt_live_slots)
        SmtLiveSlot live = array.get(smt_live_slots, slot)
        live.active := false
        live.eventId := na
        live.cycleKey := na
        array.set(smt_live_slots, slot, live)
    0

f_smt_remove_live_slot(int slot) =>
    if slot >= 0 and slot < array.size(smt_live_slots)
        SmtLiveSlot live = array.get(smt_live_slots, slot)
        if live.active and not na(live.eventId)
            int eventIdx = f_smt_find_event_by_id(live.eventId)
            if eventIdx >= 0
                f_smt_delete_event_at(eventIdx)
        f_smt_release_live_slot(slot)
    0

f_smt_finalize_live_slot(int slot, int currentCycle) =>
    bool finalized = false
    if slot >= 0 and slot < array.size(smt_live_slots)
        SmtLiveSlot live = array.get(smt_live_slots, slot)
        if live.active and not na(currentCycle) and not na(live.cycleKey) and currentCycle != live.cycleKey
            int eventIdx = f_smt_find_event_by_id(live.eventId)
            if eventIdx >= 0
                SmtEvent event = array.get(smt_events, eventIdx)
                if event.state == SMT_EVENT_LIVE
                    event.state := SMT_EVENT_CONFIRMED
                    event.projectA := false
                    event.projectB := false
                    event.bypassed := false
                    array.set(smt_events, eventIdx, event)
                    finalized := true
            f_smt_release_live_slot(slot)
    finalized

f_smt_sync_live_candidate(bool valid, int cycleKey, int previousCycleKey, int rank, int pairSlot, int dir, float y1, float y2, int t1, int t2, int b1, int b2) =>
    int slot = f_smt_live_slot(rank, pairSlot, dir)
    SmtLiveSlot live = array.get(smt_live_slots, slot)
    bool fieldsReady = not na(cycleKey) and not na(previousCycleKey) and not na(y1) and not na(y2) and not na(t1) and not na(t2) and not na(b1) and not na(b2)
    if live.active and (not valid or not fieldsReady or live.cycleKey != cycleKey)
        f_smt_remove_live_slot(slot)
        live := array.get(smt_live_slots, slot)
    if valid and fieldsReady
        int eventId = f_smt_event_id(rank, pairSlot, dir, cycleKey)
        int eventIdx = live.active ? f_smt_find_event_by_id(live.eventId) : -1
        if eventIdx < 0
            bool bodyBlocked = f_smt_body_filter_applies(rank) and f_smt_body_intersects(b1, b2, y1, y2)
            SmtEvent event = SmtEvent.new(eventId, rank, pairSlot, dir, cycleKey, previousCycleKey, y1, y2, t1, t2, b1, b2, f_smt_structure_type(dir == SMT_DIR_BULL, y1, y2), SMT_EVENT_LIVE, bodyBlocked, na, na, false, false, false)
            array.unshift(smt_events, event)
            live.active := true
            live.eventId := eventId
            live.cycleKey := cycleKey
            array.set(smt_live_slots, slot, live)
        else
            SmtEvent event = array.get(smt_events, eventIdx)
            bool pivotChanged = event.pivotAPrice != y1 or event.pivotBPrice != y2 or event.pivotATime != t1 or event.pivotBTime != t2 or event.pivotABar != b1 or event.pivotBBar != b2
            if pivotChanged
                event.pivotAPrice := y1
                event.pivotBPrice := y2
                event.pivotATime := t1
                event.pivotBTime := t2
                event.pivotABar := b1
                event.pivotBBar := b2
                event.structType := f_smt_structure_type(dir == SMT_DIR_BULL, y1, y2)
                event.bodyBlocked := f_smt_body_filter_applies(rank) and f_smt_body_intersects(b1, b2, y1, y2)
                array.set(smt_events, eventIdx, event)
    0

f_smt_event_invalidated(SmtEvent event) =>
    event.state == SMT_EVENT_CONFIRMED and bar_index > event.pivotBBar and ((event.dir == SMT_DIR_BEAR and high >= event.pivotBPrice) or (event.dir == SMT_DIR_BULL and low <= event.pivotBPrice))

f_smt_mark_historical_invalidations() =>
    int n = array.size(smt_events)
    if n > 0
        for i = n - 1 to 0
            SmtEvent event = array.get(smt_events, i)
            if f_smt_event_invalidated(event)
                event.state := SMT_EVENT_INVALIDATED
                event.projectA := false
                event.projectB := false
                array.set(smt_events, i, event)
    0

f_smt_gc_events() =>
    int i = array.size(smt_events) - 1
    while i >= 0
        if array.get(smt_events, i).state == SMT_EVENT_INVALIDATED
            f_smt_delete_event_at(i)
        i -= 1
    while array.size(smt_events) > SMT_DOMAIN_EVENT_CAP
        int removeIdx = array.size(smt_events) - 1
        while removeIdx >= 0 and array.get(smt_events, removeIdx).state == SMT_EVENT_LIVE
            removeIdx -= 1
        if removeIdx >= 0
            f_smt_delete_event_at(removeIdx)
        else
            break
    0

f_smt_event_base_render_ok(SmtEvent event) =>
    bool activeState = event.state == SMT_EVENT_LIVE or event.state == SMT_EVENT_CONFIRMED
    activeState and f_smt_rank_enabled(event.rank) and f_smt_pair_enabled(event.pairSlot) and f_smt_monthly_tf_render_ok(event.rank) and f_smt_nearby_render_ok(event.pivotBBar) and f_smt_render_time_ok(event.pivotBTime) and not event.bodyBlocked

f_smt_hard_winner(SmtEvent dominant, SmtEvent candidate) =>
    dominant.rank > candidate.rank or (dominant.rank == candidate.rank and dominant.pairSlot < candidate.pairSlot) or (dominant.rank == candidate.rank and dominant.pairSlot == candidate.pairSlot and dominant.id < candidate.id)

f_smt_hard_loses(int idx) =>
    bool loses = false
    SmtEvent candidate = array.get(smt_events, idx)
    int n = array.size(smt_events)
    if candidate.projectA and candidate.dir != 0 and not na(candidate.pivotATime) and not na(candidate.pivotBTime) and n > 1
        for i = 0 to n - 1
            if i != idx and not loses
                SmtEvent other = array.get(smt_events, i)
                bool sameEvent = other.projectA and other.dir == candidate.dir and other.pivotATime == candidate.pivotATime and other.pivotBTime == candidate.pivotBTime
                loses := sameEvent and f_smt_hard_winner(other, candidate)
    loses

f_smt_inside_dominates(SmtEvent keeper, SmtEvent candidate) =>
    int keepLeft = math.min(keeper.pivotABar, keeper.pivotBBar)
    int keepRight = math.max(keeper.pivotABar, keeper.pivotBBar)
    int candidateLeft = math.min(candidate.pivotABar, candidate.pivotBBar)
    int candidateRight = math.max(candidate.pivotABar, candidate.pivotBBar)
    int keepSpan = keepRight - keepLeft
    int candidateSpan = candidateRight - candidateLeft
    bool directionOk = keeper.dir != 0 and keeper.dir == candidate.dir
    bool structureOk = f_smt_cleanup_struct_ok(candidate.rank, candidate.structType, keeper.structType)
    bool priorityOk = keeper.rank > candidate.rank or (keeper.rank == candidate.rank and keepSpan > candidateSpan)
    candidateLeft >= keepLeft + 1 and candidateRight <= keepRight - 1 and keepSpan > candidateSpan and candidateSpan > 0 and keepSpan >= 3 and directionOk and structureOk and priorityOk

f_smt_inside_loses(int idx) =>
    bool loses = false
    SmtEvent candidate = array.get(smt_events, idx)
    int n = array.size(smt_events)
    if candidate.projectB and n > 1
        for i = 0 to n - 1
            if i != idx and not loses
                SmtEvent keeper = array.get(smt_events, i)
                loses := keeper.projectB and f_smt_inside_dominates(keeper, candidate)
    loses

f_smt_pivot_winner(SmtEvent dominant, SmtEvent candidate) =>
    dominant.rank > candidate.rank or (dominant.rank == candidate.rank and dominant.id < candidate.id)

f_smt_pivot_loses(int idx, bool checkPivotA) =>
    bool loses = false
    SmtEvent candidate = array.get(smt_events, idx)
    int n = array.size(smt_events)
    if candidate.projectA and candidate.dir != 0 and n > 1
        for i = 0 to n - 1
            if i != idx and not loses
                SmtEvent other = array.get(smt_events, i)
                bool samePivot = checkPivotA ? other.pivotATime == candidate.pivotATime : other.pivotBTime == candidate.pivotBTime
                bool directionOk = other.dir == candidate.dir
                bool structureOk = f_smt_cleanup_struct_ok(candidate.rank, candidate.structType, other.structType)
                loses := other.projectA and samePivot and directionOk and structureOk and f_smt_pivot_winner(other, candidate)
    loses

f_smt_prepare_projection() =>
    int n = array.size(smt_events)
    if n > 0
        for i = 0 to n - 1
            SmtEvent event = array.get(smt_events, i)
            bool baseEligible = f_smt_event_base_render_ok(event)
            event.projectA := baseEligible
            event.projectB := baseEligible
            event.bypassed := false
            array.set(smt_events, i, event)

        // Exact A-B arbitration is independent from optional cleanup modes.
        for i = 0 to n - 1
            SmtEvent event = array.get(smt_events, i)
            event.projectB := event.projectA and not f_smt_hard_loses(i)
            array.set(smt_events, i, event)

        bool insideBypassMode = smt_cleanup_mode == "Bypass Inside Range"
        bool evaluateInside = smt_cleanup_inside or insideBypassMode
        for i = 0 to n - 1
            SmtEvent event = array.get(smt_events, i)
            bool losesInside = evaluateInside and event.projectB and f_smt_inside_loses(i)
            event.projectA := event.projectB and not (smt_cleanup_inside and losesInside)
            event.bypassed := smt_aesthetic_bypass and insideBypassMode and losesInside
            array.set(smt_events, i, event)

        bool pivotABypassMode = smt_cleanup_mode == "Bypass Pivot A"
        bool pivotBBypassMode = smt_cleanup_mode == "Bypass Pivot B"
        bool evaluatePivotA = smt_cleanup_pivot_a or pivotABypassMode
        bool evaluatePivotB = smt_cleanup_pivot_b or pivotBBypassMode
        for i = 0 to n - 1
            SmtEvent event = array.get(smt_events, i)
            bool losesPivotA = evaluatePivotA and event.projectA and f_smt_pivot_loses(i, true)
            bool losesPivotB = evaluatePivotB and event.projectA and f_smt_pivot_loses(i, false)
            bool removed = (smt_cleanup_pivot_a and losesPivotA) or (smt_cleanup_pivot_b and losesPivotB)
            event.projectB := event.projectA and not removed
            bool pivotBypassed = (pivotABypassMode and losesPivotA) or (pivotBBypassMode and losesPivotB)
            event.bypassed := event.bypassed or (smt_aesthetic_bypass and pivotBypassed)
            array.set(smt_events, i, event)
    0

f_smt_sync_render_at(int idx, bool shouldRender) =>
    if idx >= 0 and idx < array.size(smt_events)
        SmtEvent event = array.get(smt_events, idx)
        if shouldRender
            color eventColor = event.bypassed ? color.new(f_smt_rank_color(event.rank), 65) : f_smt_rank_color(event.rank)
            if na(event.lineId)
                event.lineId := line.new(x1=event.pivotATime, y1=event.pivotAPrice, x2=event.pivotBTime, y2=event.pivotBPrice, xloc=xloc.bar_time, color=eventColor, width=1, style=smt_line_style)
            else
                line.set_xy1(event.lineId, event.pivotATime, event.pivotAPrice)
                line.set_xy2(event.lineId, event.pivotBTime, event.pivotBPrice)
                line.set_color(event.lineId, eventColor)
                line.set_style(event.lineId, smt_line_style)
            bool labelsOn = f_smt_labels_for_rank(event.rank)
            if labelsOn
                int labelTime = f_smt_event_label_time(event.pivotATime, event.pivotBTime, event.pivotABar, event.pivotBBar)
                string defaultText = f_smt_default_label_text(event.rank, event.pairSlot)
                string displayText = event.bypassed ? (smt_bypass_labels_show ? f_smt_bypass_label_text(event.rank, event.pairSlot) : "") : defaultText
                color textColor = event.bypassed ? (smt_bypass_labels_show ? color.new(smt_txt_all, 50) : color.new(smt_txt_all, 100)) : smt_txt_all
                if na(event.labelId)
                    event.labelId := label.new(x=labelTime, y=f_smt_event_mid_y(event.pivotAPrice, event.pivotBPrice), text=displayText, xloc=xloc.bar_time, yloc=yloc.price, color=color.new(color.white, 100), style=event.dir == SMT_DIR_BULL ? label.style_label_up : label.style_label_down, textcolor=textColor, size=smt_label_size, tooltip=defaultText)
                else
                    label.set_x(event.labelId, labelTime)
                    label.set_y(event.labelId, f_smt_event_mid_y(event.pivotAPrice, event.pivotBPrice))
                    label.set_text(event.labelId, displayText)
                    label.set_textcolor(event.labelId, textColor)
                    label.set_size(event.labelId, smt_label_size)
            else if not na(event.labelId)
                label.delete(event.labelId)
                event.labelId := na
        else
            if not na(event.labelId)
                label.delete(event.labelId)
                event.labelId := na
            if not na(event.lineId)
                line.delete(event.lineId)
                event.lineId := na
        array.set(smt_events, idx, event)
    0

f_smt_project_events() =>
    if barstate.islast
        f_smt_prepare_projection()
        array<int> bucketCounts = array.new_int(SMT_LIVE_SLOTS, 0)
        int rendered = 0
        int n = array.size(smt_events)
        if n > 0
            // Global object pressure is resolved by rank first, then recency.
            for rankCursor = SMT_STATE_CYCLES to 1
                for i = 0 to n - 1
                    SmtEvent event = array.get(smt_events, i)
                    if event.rank == rankCursor
                        bool shouldRender = event.projectB
                        if shouldRender and event.state == SMT_EVENT_CONFIRMED
                            int bucket = f_smt_live_slot(event.rank, event.pairSlot, event.dir)
                            int bucketCount = array.get(bucketCounts, bucket)
                            shouldRender := bucketCount < smt_history_limit
                            if shouldRender
                                array.set(bucketCounts, bucket, bucketCount + 1)
                        shouldRender := shouldRender and rendered < SMT_RENDER_EVENT_CAP
                        if shouldRender
                            rendered += 1
                        event.projectA := shouldRender
                        array.set(smt_events, i, event)
            for i = 0 to n - 1
                f_smt_sync_render_at(i, array.get(smt_events, i).projectA)
    0

f_smt_run_pair(SmtRangeState mainState, SmtRangeState pairState, bool engineEnabled, SmtCorrelationDescriptor pair, int rank, int pairSlot) =>
    if not na(mainState.cycleKey)
        int bearSlot = f_smt_live_slot(rank, pairSlot, SMT_DIR_BEAR)
        int bullSlot = f_smt_live_slot(rank, pairSlot, SMT_DIR_BULL)
        // Finalization belongs to the main range transition and never to a line.
        f_smt_finalize_live_slot(bearSlot, mainState.cycleKey)
        f_smt_finalize_live_slot(bullSlot, mainState.cycleKey)
        bool cycleAligned = mainState.cycleKey == pairState.cycleKey and not na(mainState.previousCycleKey) and mainState.previousCycleKey == pairState.previousCycleKey
        bool rangesReady = mainState.hasCurrentRange and mainState.hasPreviousRange and pairState.hasCurrentRange and pairState.hasPreviousRange
        bool canResolve = engineEnabled and pair.configured and pair.dataReady and cycleAligned and rangesReady
        if canResolve
            SmtSweepState mainSweep = f_smt_build_sweeps(mainState, false)
            SmtSweepState pairSweep = f_smt_build_sweeps(pairState, pair.inverse)
            [bearValid, bullValid] = f_smt_resolve_pair_state(mainSweep, pairSweep)
            f_smt_sync_live_candidate(bearValid, mainState.cycleKey, mainState.previousCycleKey, rank, pairSlot, SMT_DIR_BEAR, mainSweep.highPivotA, mainSweep.highPivotB, mainSweep.highTimeA, mainSweep.highTimeB, mainSweep.highBarA, mainSweep.highBarB)
            f_smt_sync_live_candidate(bullValid, mainState.cycleKey, mainState.previousCycleKey, rank, pairSlot, SMT_DIR_BULL, mainSweep.lowPivotA, mainSweep.lowPivotB, mainSweep.lowTimeA, mainSweep.lowTimeB, mainSweep.lowBarA, mainSweep.lowBarB)
        else
            f_smt_sync_live_candidate(false, mainState.cycleKey, mainState.previousCycleKey, rank, pairSlot, SMT_DIR_BEAR, na, na, na, na, na, na)
            f_smt_sync_live_candidate(false, mainState.cycleKey, mainState.previousCycleKey, rank, pairSlot, SMT_DIR_BULL, na, na, na, na, na, na)
    0

f_run_smt_cycle(int cycleIdx, bool engineEnabled, int rangeKey, int rank) =>
    // Calculation gates own engine cost. Show toggles own only final projection.
    f_smt_update_state(cycleIdx, SMT_SRC_MAIN, engineEnabled, rangeKey, high, low, time)
    f_smt_update_state(cycleIdx, SMT_SRC_PAIR1, engineEnabled and smt_corr_pair1.configured, rangeKey, smt_corr_pair1.sourceHigh, smt_corr_pair1.sourceLow, smt_corr_pair1.sourceTime)
    f_smt_update_state(cycleIdx, SMT_SRC_PAIR2, engineEnabled and smt_corr_pair2.configured, rangeKey, smt_corr_pair2.sourceHigh, smt_corr_pair2.sourceLow, smt_corr_pair2.sourceTime)
    f_smt_update_state(cycleIdx, SMT_SRC_PAIR3, engineEnabled and smt_corr_pair3.configured, rangeKey, smt_corr_pair3.sourceHigh, smt_corr_pair3.sourceLow, smt_corr_pair3.sourceTime)
    SmtRangeState mainState = f_smt_get_state(cycleIdx, SMT_SRC_MAIN)
    SmtRangeState pair1State = f_smt_get_state(cycleIdx, SMT_SRC_PAIR1)
    SmtRangeState pair2State = f_smt_get_state(cycleIdx, SMT_SRC_PAIR2)
    SmtRangeState pair3State = f_smt_get_state(cycleIdx, SMT_SRC_PAIR3)
    f_smt_run_pair(mainState, pair1State, engineEnabled, smt_corr_pair1, rank, 1)
    f_smt_run_pair(mainState, pair2State, engineEnabled, smt_corr_pair2, rank, 2)
    f_smt_run_pair(mainState, pair3State, engineEnabled, smt_corr_pair3, rank, 3)
    0
