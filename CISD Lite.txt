cisd_render_active_line(line l, int anchor_t, int stop_t, float lvl, color clr) =>
    out = l
    if not na(anchor_t) and not na(stop_t) and not na(lvl)
        if na(out)
            out := line.new(x1=anchor_t, y1=lvl, x2=stop_t, y2=lvl, xloc=xloc.bar_time, color=clr, width=cisd_line_width)
        else
            line.set_xy1(out, anchor_t, lvl)
            line.set_xy2(out, stop_t, lvl)
            line.set_color(out, clr)
            line.set_width(out, cisd_line_width)
    out
cisd_safe_line_delete(line l) =>
    if not na(l)
        line.delete(l)
htf_cisd_manage_active() =>
    dirty = false
    n = cisd_event_registry.size()
    if n > 0
        for i = n - 1 to 0
            ev = cisd_event_registry.get(i)
            l = ev.ln
            lvl = ev.lvl
            dir = ev.dir
            conf_t = ev.conf_t
            use_inv = na(ev.inv_lvl) ? lvl : ev.inv_lvl
            visible_clr = ev.disp_ok ? (dir == 1 ? cisd_bull_clr : cisd_bear_clr) : #FFFFFF00
            live_now = QTU.cisdInvalidLive(conf_t, time)
            formed_bar = nz(ev.formed_bar, ev.conf_bar)
            inv_eligible = not na(formed_bar) and bar_index > formed_bar
            if barstate.isconfirmed
                if inv_eligible and QTU.cisdPostConfirmInvalid(dir, use_inv, ev.c2_ext, formed_bar, HTF_CISD_OFFSET_CAP, bar_index, RECENT_BAR_WINDOW, barstate.isconfirmed)
                    cisd_safe_line_delete(l)
                    cisd_event_registry.remove(i)
                    dirty := true
                else
                    if not ev.disp_ok and not na(ev.conf_bar) and not na(ev.disp_lvl) and bar_index > ev.conf_bar
                        ev.disp_ok := dir == 1 ? close > ev.disp_lvl : close < ev.disp_lvl
                        dirty := dirty or ev.disp_ok
                    stale_no_disp = not ev.disp_ok and not na(ev.conf_bar) and bar_index - ev.conf_bar >= 50
                    if stale_no_disp
                        cisd_safe_line_delete(l)
                        cisd_event_registry.remove(i)
                        dirty := true
                    else
                        if ev.pending_inv
                            ev.pending_inv := false
                            dirty := true
                        next_stop = QTU.cisdNextStopTime(ev.stop_t, ev.anchor_t, lvl, int(time), int(time[1]), open, close, barstate.isconfirmed)
                        dirty := dirty or next_stop != ev.stop_t
                        ev.stop_t := next_stop
                        ev.ln := cisd_render_active_line(l, ev.anchor_t, ev.stop_t, lvl, visible_clr)
                        cisd_event_registry.set(i, ev)
            else
                pending_now = inv_eligible and live_now and ((dir == 1 ? low < use_inv : dir == -1 ? high > use_inv : false) or (not na(ev.c2_ext) and QTU.cisdC2Invalidates(dir, ev.c2_ext, low, high)))
                if pending_now
                    if not ev.pending_inv
                        ev.pending_inv := true
                        dirty := true
                    ev.ln := cisd_render_active_line(l, ev.anchor_t, ev.stop_t, lvl, #FFFFFF00)
                    cisd_event_registry.set(i, ev)
                else
                    if ev.pending_inv
                        ev.pending_inv := false
                        dirty := true
                    ev.ln := cisd_render_active_line(l, ev.anchor_t, ev.stop_t, lvl, visible_clr)
                    cisd_event_registry.set(i, ev)
    dirty
cisd_event_pop_oldest_cycle() =>
    if cisd_event_registry.size() > 0
        old = cisd_event_registry.pop()
        cisd_safe_line_delete(old.ln)
cisd_event_cleanup_state_cap() =>
    while cisd_event_registry.size() > CISD_EVENT_STATE_CAP
        cisd_event_pop_oldest_cycle()
    0
cisd_render_display_cap() =>
    displayed = 0
    i = 0
    while i < cisd_event_registry.size()
        ev = cisd_event_registry.get(i)
        if not na(ev.ln)
            if ev.disp_ok and not ev.pending_inv
                displayed += 1
                base_clr = ev.dir == 1 ? cisd_bull_clr : cisd_bear_clr
                line.set_color(ev.ln, displayed <= cisd_display_cap ? base_clr : #FFFFFF00)
            else
                line.set_color(ev.ln, #FFFFFF00)
        i += 1
    0
cisd_event_clear_all() =>
    while cisd_event_registry.size() > 0
        cisd_event_pop_oldest_cycle()
    0

htf_cisd_update_active_lines(bool _runtime_on) =>
    if not cisd_show
        if cisd_event_registry.size() > 0
            cisd_event_clear_all()
    else
        if _runtime_on and cisd_event_registry.size() > 0
            htf_cisd_manage_active()
        if cisd_event_registry.size() > 0
            cisd_render_display_cap()
    0

method reset_draw_refs(array<UDT_HTF_Candle> candles) =>
    for candle in candles
        candle.low_line := na
        candle.high_line := na
        for sweep in candle.ltf_sweeps
            sweep.ltf_line := na
            sweep.htf_line := na
        for sweep in candle.htf_sweeps
            sweep.ltf_line := na
            sweep.htf_line := na
method clean_bin(UDT_Store store) =>
    for obj in store.bin_ln
        obj.delete()
    for obj in store.bin_box
        obj.delete()
    for obj in store.bin_lbl
        obj.delete()
    store.bin_ln.clear()
    store.bin_box.clear()
    store.bin_lbl.clear()
    htf_1_candles.reset_draw_refs()
    htf_2_candles.reset_draw_refs()
    htf_3_candles.reset_draw_refs()


htf_live_eval_ok() =>
    barstate.isrealtime or barstate.isconfirmed
htf_cisd_sweep_permanent_invalid(UDT_Sweep _sw) =>
    not na(_sw.invalidated_on)
htf_cisd_sweep_active_valid(UDT_Sweep _sw) =>
    inside_side = not na(_sw.y) and (_sw.bull ? close <= _sw.y : close >= _sw.y)
    not na(_sw.y) and not htf_cisd_sweep_permanent_invalid(_sw) and inside_side
method cisd_reject(UDT_Sweep sw, bool full) =>
    sw.cisd_state := 3
    if full
        sw.cisd_dir := 0
        sw.cisd_c2_bar := na
        sw.cisd_c2_ext := na
        sw.cisd_htf_c2_idx := na
    sw.cisd_lvl := na
    sw.cisd_inv_lvl := na
    sw.cisd_set_high := na
    sw.cisd_set_low := na
    sw.cisd_set_wick_high := na
    sw.cisd_set_wick_low := na
    sw.cisd_anchor_bar := na
    sw.cisd_anchor_t := na
    sw.cisd_conf_bar := na
    sw.cisd_conf_t := na
    sw
htf_cisd_prune_consumed() =>
    min_bar = QTU.cisdRecentMinBar(bar_index, RECENT_BAR_WINDOW)
    n = cisd_consumed.size()
    if n > 0
        for i = n - 1 to 0
            if cisd_consumed.get(i).bar < min_bar
                cisd_consumed.remove(i)
    while cisd_consumed.size() > HTF_CISD_CONSUMED_MAX
        cisd_consumed.remove(0)
    0
htf_cisd_active_event_exists(int _dir, int _c2_bar, int _pane) =>
    found = false
    n = cisd_event_registry.size()
    if n > 0
        for i = n - 1 to 0
            ev = cisd_event_registry.get(i)
            if ev.dir == _dir and ev.c2_bar == _c2_bar and ev.p == _pane
                found := true
                break
    found
htf_cisd_event_consumed(int _dir, int _c2_bar, int _pane, int _src_x1, float _src_y) =>
    found = false
    n = cisd_consumed.size()
    if _dir != 0 and not na(_c2_bar) and n > 0
        for i = n - 1 to 0
            k = cisd_consumed.get(i)
            if k.dir == _dir and k.c2 == _c2_bar and k.pane == _pane and k.x1 == _src_x1 and QTU.cisdPriceMatch(k.y, _src_y, syminfo.mintick)
                found := true
                break
    found and htf_cisd_active_event_exists(_dir, _c2_bar, _pane)
htf_cisd_mark_event_consumed(int _dir, int _c2_bar, int _pane, int _src_x1, float _src_y) =>
    if not htf_cisd_event_consumed(_dir, _c2_bar, _pane, _src_x1, _src_y)
        cisd_consumed.push(CisdConsumed.new(_dir, _c2_bar, _pane, _src_x1, _src_y, bar_index))
    htf_cisd_prune_consumed()
    0
htf_cisd_source_dead_before_bar(UDT_Sweep _sw, int _bar) =>
    not na(_sw.invalidated_on) and (na(_bar) or _sw.invalidated_on <= _bar)
htf_cisd_source_dead_before_confirm(UDT_Sweep _sw) =>
    htf_cisd_source_dead_before_bar(_sw, _sw.cisd_conf_bar)
htf_cisd_c2_extreme_branched(int _dir, int _c2_bar, float _c2_ext) =>
    branched = false
    if _dir != 0 and not na(_c2_bar) and not na(_c2_ext)
        span = math.min(bar_index - _c2_bar, RECENT_BAR_WINDOW)
        if span > 0
            for off = 0 to span - 1
                if bar_index - off > _c2_bar
                    if _dir == 1 ? low[off] < _c2_ext : high[off] > _c2_ext
                        branched := true
                        break
    branched
htf_cisd_c2_extreme_branched_in_pane(array<UDT_HTF_Candle> _candles, int _dir, int _c2_bar, float _c2_ext) =>
    branched = false
    if _dir != 0 and not na(_c2_bar) and not na(_c2_ext) and array.size(_candles) > 0
        for c in _candles
            if c.c_idx > _c2_bar
                if _dir == 1 ? c.l < _c2_ext : c.h > _c2_ext
                    branched := true
                    break
    branched
htf_cisd_inv_level_veto(int _dir, float _inv_lvl, int _cisd_anchor_bar) =>
    // Reuse the confirmed-bar invalidation scan with the CISD anchor as its origin.
    // cisdPostConfirmInvalid begins at origin + 1, so the anchor candle itself never vetoes the setup.
    _dir != 0 and not na(_inv_lvl) and not na(_cisd_anchor_bar) and QTU.cisdPostConfirmInvalid(_dir, _inv_lvl, na, _cisd_anchor_bar, HTF_CISD_OFFSET_CAP, bar_index, RECENT_BAR_WINDOW, barstate.isconfirmed)

htf_cisd_candidate_better(HtfCisdCandidate _candidate, HtfCisdCandidate _current) =>
    better = _candidate.ok and not _current.ok
    if _candidate.ok and _current.ok
        cb = nz(_candidate.c2_bar, -1)
        bb = nz(_current.c2_bar, -1)
        better := cb > bb
        if not better and cb == bb
            ch = nz(_candidate.htf_c2_idx, -1)
            bh = nz(_current.htf_c2_idx, -1)
            better := ch > bh
    better

f_cisd_set_ranges_from_locked_c2(int _c2_bar, int _dir) =>
    float set_high = na
    float set_low = na
    float set_wick_high = na
    float set_wick_low = na
    int anchor_bar = na
    if not na(_c2_bar)
        scan_left_bar = math.max(QTU.cisdRecentMinBar(bar_index, RECENT_BAR_WINDOW), _c2_bar - HTF_CISD_CORE_SCANBACK_BARS)
        off_c2 = QTU.cisdAbsBarToOff(_c2_bar, CISD_CORE_SCAN_CAP, bar_index, RECENT_BAR_WINDOW)
        off_left = QTU.cisdAbsBarToOff(scan_left_bar, CISD_CORE_SCAN_CAP, bar_index, RECENT_BAR_WINDOW)
        if not na(off_c2) and not na(off_left) and off_left >= off_c2
            scan_upclose = _dir == -1
            max_candles = math.max(1, cisd_max_candles)
            max_k = math.min(off_left - off_c2, CISD_CORE_SCAN_CAP - off_c2)
            c2_upclose = close[off_c2] > open[off_c2]
            c2_downclose = close[off_c2] < open[off_c2]
            start_k = (scan_upclose ? c2_upclose : c2_downclose) ? 0 : 1
            count = 0
            started = false
            if start_k <= max_k
                for k = 0 to CISD_CORE_SCAN_CAP
                    if k > max_k
                        break
                    if k >= start_k
                        idx = off_c2 + k
                        if idx > CISD_CORE_SCAN_CAP or na(time[idx])
                            break
                        upclose = close[idx] > open[idx]
                        downclose = close[idx] < open[idx]
                        accepted = scan_upclose ? upclose : downclose
                        opposite = scan_upclose ? downclose : upclose
                        if accepted
                            started := true
                            count += 1
                            high_endpoint = _dir == 1 ? open[idx] : close[idx]
                            low_endpoint = _dir == 1 ? close[idx] : open[idx]
                            set_high := na(set_high) ? high_endpoint : math.max(set_high, high_endpoint)
                            set_low := na(set_low) ? low_endpoint : math.min(set_low, low_endpoint)
                            set_wick_high := na(set_wick_high) ? high[idx] : math.max(set_wick_high, high[idx])
                            set_wick_low := na(set_wick_low) ? low[idx] : math.min(set_wick_low, low[idx])
                            anchor_bar := bar_index - idx
                            if count == max_candles
                                break
                        else if started and opposite
                            break
    [set_high, set_low, set_wick_high, set_wick_low, anchor_bar]

htf_cisd_maintain_pane(array<UDT_HTF_Candle> _candles, int _pane, bool _enabled) =>
    ready = false
    count = _candles.size()
    if _enabled and count > 1
        max_i = math.min(count - 1, HTF_SWEEP_STATE_CANDLES - 1)
        if max_i >= 1
            for i = max_i to 1 by 1
                c1 = _candles.get(i)
                c2 = _candles.get(i - 1)
                sn = c1.ltf_sweeps.size()
                if sn > 0
                    set_c1 = false
                    for j = 0 to sn - 1
                        sw = c1.ltf_sweeps.get(j)
                        dir = sw.bull ? -1 : 1
                        changed = false
                        sweep_valid_for_cisd = htf_cisd_sweep_active_valid(sw)
                        if sw.cisd_state == 0 or sw.cisd_state == 1 or sw.cisd_state == 2
                            if sweep_valid_for_cisd and c2.is_closed
                                c2_extreme = dir == 1 ? c2.l : c2.h
								[locked_bar, locked_time, locked_ext] = QTU.cisdFindLockedC2(sw.c2_left, sw.c2_right, dir, c2_extreme, HTF_CISD_OFFSET_CAP, bar_index, RECENT_BAR_WINDOW, barstate.isconfirmed, syminfo.mintick)
                                if not na(locked_bar) and not na(locked_time) and not na(locked_ext)
                                    lock_changed = sw.cisd_state == 0 or sw.cisd_dir != dir or sw.cisd_c2_bar != locked_bar or not QTU.cisdPriceMatch(sw.cisd_c2_ext, locked_ext, syminfo.mintick)
                                    if lock_changed
                                        sw.cisd_state := 1
                                        sw.cisd_dir := dir
                                        sw.cisd_c2_bar := locked_bar
                                        sw.cisd_c2_ext := locked_ext
                                        sw.cisd_htf_c2_idx := c2.c_idx
                                        sw.cisd_lvl := na
                                        sw.cisd_inv_lvl := na
                                        sw.cisd_set_high := na
                                        sw.cisd_set_low := na
                                        sw.cisd_set_wick_high := na
                                        sw.cisd_set_wick_low := na
                                        sw.cisd_anchor_bar := na
                                        sw.cisd_anchor_t := na
                                        sw.cisd_conf_bar := na
                                        sw.cisd_conf_t := na
                                        changed := true
                                    if sw.cisd_state == 1
                                        [core_lvl, core_t, inv_lvl] = QTU.cisdPickCoreFromLockedC2(sw.cisd_c2_bar, sw.cisd_dir, HTF_CISD_CORE_SCANBACK_BARS, CISD_CORE_SCAN_CAP, bar_index, RECENT_BAR_WINDOW, cisd_max_candles)
                                        [set_high, set_low, set_wick_high, set_wick_low, anchor_bar] = f_cisd_set_ranges_from_locked_c2(sw.cisd_c2_bar, sw.cisd_dir)
                                        core_changed = not QTU.cisdPriceMatch(sw.cisd_lvl, core_lvl, syminfo.mintick) or not QTU.cisdPriceMatch(sw.cisd_inv_lvl, inv_lvl, syminfo.mintick) or not QTU.cisdPriceMatch(sw.cisd_set_high, set_high, syminfo.mintick) or not QTU.cisdPriceMatch(sw.cisd_set_low, set_low, syminfo.mintick) or not QTU.cisdPriceMatch(sw.cisd_set_wick_high, set_wick_high, syminfo.mintick) or not QTU.cisdPriceMatch(sw.cisd_set_wick_low, set_wick_low, syminfo.mintick) or sw.cisd_anchor_bar != anchor_bar or sw.cisd_anchor_t != core_t
                                        if not na(core_lvl) and not na(core_t) and not na(inv_lvl) and not na(set_high) and not na(set_low) and not na(set_wick_high) and not na(set_wick_low) and not na(anchor_bar)
                                            if core_changed
                                                sw.cisd_lvl := core_lvl
                                                sw.cisd_inv_lvl := inv_lvl
                                                sw.cisd_set_high := set_high
                                                sw.cisd_set_low := set_low
                                                sw.cisd_set_wick_high := set_wick_high
                                                sw.cisd_set_wick_low := set_wick_low
                                                sw.cisd_anchor_bar := anchor_bar
                                                sw.cisd_anchor_t := core_t
                                                sw.cisd_conf_bar := na
                                                sw.cisd_conf_t := na
                                                changed := true
                                            if na(sw.cisd_conf_bar)
                                                [conf_bar, conf_t] = QTU.cisdFindConfirmAfterC2(sw.cisd_c2_bar, sw.cisd_lvl, sw.cisd_dir, HTF_CISD_CONFIRM_MAX_BARS, HTF_CISD_OFFSET_CAP, bar_index, RECENT_BAR_WINDOW, barstate.isconfirmed)
                                                if not na(conf_bar) and not na(conf_t)
                                                    sw.cisd_conf_bar := conf_bar
                                                    sw.cisd_conf_t := conf_t
                                                    changed := true
                                        else
                                            if not na(sw.cisd_lvl) or not na(sw.cisd_inv_lvl) or not na(sw.cisd_set_high) or not na(sw.cisd_set_low) or not na(sw.cisd_set_wick_high) or not na(sw.cisd_set_wick_low) or not na(sw.cisd_anchor_bar) or not na(sw.cisd_anchor_t) or not na(sw.cisd_conf_bar) or not na(sw.cisd_conf_t)
                                                sw.cisd_lvl := na
                                                sw.cisd_inv_lvl := na
                                                sw.cisd_set_high := na
                                                sw.cisd_set_low := na
                                                sw.cisd_set_wick_high := na
                                                sw.cisd_set_wick_low := na
                                                sw.cisd_anchor_bar := na
                                                sw.cisd_anchor_t := na
                                                sw.cisd_conf_bar := na
                                                sw.cisd_conf_t := na
                                                changed := true
                                        source_dead_before_conf = htf_cisd_source_dead_before_confirm(sw)
                                        if source_dead_before_conf
                                            sw := sw.cisd_reject(false)
                                            changed := true
                            else if sw.cisd_state == 1 and htf_cisd_source_dead_before_confirm(sw)
                                sw := sw.cisd_reject(true)
                                changed := true
                        consumed_active = htf_cisd_event_consumed(sw.cisd_dir, sw.cisd_c2_bar, _pane, sw.x1, sw.y)
                        if sw.cisd_state == 1 and consumed_active
                            sw.cisd_state := 2
                            changed := true
                        else if sw.cisd_state == 2 and not consumed_active and not htf_cisd_source_dead_before_confirm(sw) and not htf_cisd_c2_extreme_branched(sw.cisd_dir, sw.cisd_c2_bar, sw.cisd_c2_ext) and not htf_cisd_c2_extreme_branched_in_pane(_candles, sw.cisd_dir, sw.cisd_c2_bar, sw.cisd_c2_ext)
                            sw.cisd_state := 1
                            changed := true
                        ready := ready or sw.cisd_state == 1
                        if changed
                            c1.ltf_sweeps.set(j, sw)
                            set_c1 := true
                    if set_c1
                        _candles.set(i, c1)
    ready
htf_cisd_select_pane(array<UDT_HTF_Candle> _candles, int _pane, bool _enabled, HtfCisdCandidate _best) =>
    best = _best
    count = _candles.size()
    if _enabled and count > 1
        max_i = math.min(count - 1, HTF_SWEEP_STATE_CANDLES - 1)
        if max_i >= 1
            for i = max_i to 1 by 1
                c1 = _candles.get(i)
                sn = c1.ltf_sweeps.size()
                if sn > 0
                    set_c1 = false
                    for j = 0 to sn - 1
                        sw = c1.ltf_sweeps.get(j)
                        changed = false
                        if sw.cisd_state == 1
                            consumed = htf_cisd_event_consumed(sw.cisd_dir, sw.cisd_c2_bar, _pane, sw.x1, sw.y)
                            source_dead = htf_cisd_source_dead_before_confirm(sw)
                            if consumed
                                sw.cisd_state := 2
                                changed := true
                            else if source_dead
                                sw := sw.cisd_reject(false)
                                changed := true
                            else if htf_cisd_c2_extreme_branched(sw.cisd_dir, sw.cisd_c2_bar, sw.cisd_c2_ext) or htf_cisd_c2_extreme_branched_in_pane(_candles, sw.cisd_dir, sw.cisd_c2_bar, sw.cisd_c2_ext)
                                sw := sw.cisd_reject(true)
                                changed := true
                            else if htf_cisd_inv_level_veto(sw.cisd_dir, sw.cisd_inv_lvl, sw.cisd_anchor_bar)
                                sw := sw.cisd_reject(false)
                                changed := true
                            else if not na(sw.cisd_lvl) and not na(sw.cisd_inv_lvl) and not na(sw.cisd_anchor_t) and not na(sw.cisd_conf_bar) and sw.cisd_conf_bar <= bar_index
                                cand = HtfCisdCandidate.new()
                                cand.ok := true
                                cand.pane := _pane
                                cand.dir := sw.cisd_dir
                                cand.htf_c2_idx := sw.cisd_htf_c2_idx
                                cand.c2_bar := sw.cisd_c2_bar
                                cand.c2_ext := sw.cisd_c2_ext
                                cand.cisd_lvl := sw.cisd_lvl
                                cand.cisd_inv_lvl := sw.cisd_inv_lvl
                                cand.cisd_set_high := sw.cisd_set_high
                                cand.cisd_set_low := sw.cisd_set_low
                                cand.cisd_set_wick_high := sw.cisd_set_wick_high
                                cand.cisd_set_wick_low := sw.cisd_set_wick_low
                                cand.cisd_anchor_bar := sw.cisd_anchor_bar
                                cand.anchor_t := sw.cisd_anchor_t
                                cand.conf_bar := sw.cisd_conf_bar
                                cand.conf_t := sw.cisd_conf_t
                                cand.disp_lvl := QTU.cisdDispLevel(sw.cisd_dir, sw.cisd_conf_bar, HTF_CISD_OFFSET_CAP, bar_index, RECENT_BAR_WINDOW)
                                cand.disp_ok := QTU.cisdDispSeen(sw.cisd_dir, sw.cisd_conf_bar, cand.disp_lvl, HTF_CISD_OFFSET_CAP, bar_index, RECENT_BAR_WINDOW, barstate.isconfirmed, 50)
                                cand.src_x1 := sw.x1
                                cand.src_y := sw.y
                                if htf_cisd_candidate_better(cand, best)
                                    best := cand
                        if changed
                            c1.ltf_sweeps.set(j, sw)
                            set_c1 := true
                    if set_c1
                        _candles.set(i, c1)
    best
htf_cisd_select_early_pane(array<UDT_HTF_Candle> _candles, int _pane, bool _enabled, HtfCisdCandidate _best) =>
    best = _best
    count = _candles.size()
    if _enabled and count > 1 and not barstate.isconfirmed
        max_i = math.min(count - 1, HTF_SWEEP_STATE_CANDLES - 1)
        if max_i >= 1
            for i = max_i to 1 by 1
                c1 = _candles.get(i)
                sn = c1.ltf_sweeps.size()
                if sn > 0
                    for j = 0 to sn - 1
                        sw = c1.ltf_sweeps.get(j)
                        if sw.cisd_state == 1 and na(sw.cisd_conf_bar) and not na(sw.cisd_lvl) and not na(sw.cisd_inv_lvl) and not na(sw.cisd_anchor_t) and not na(sw.cisd_c2_bar)
                            consumed = htf_cisd_event_consumed(sw.cisd_dir, sw.cisd_c2_bar, _pane, sw.x1, sw.y)
                            source_dead = htf_cisd_source_dead_before_confirm(sw)
                            extreme_branched = htf_cisd_c2_extreme_branched(sw.cisd_dir, sw.cisd_c2_bar, sw.cisd_c2_ext) or htf_cisd_c2_extreme_branched_in_pane(_candles, sw.cisd_dir, sw.cisd_c2_bar, sw.cisd_c2_ext)
                            inv_vetoed = htf_cisd_inv_level_veto(sw.cisd_dir, sw.cisd_inv_lvl, sw.cisd_anchor_bar)
                            early_cross = bar_index > sw.cisd_c2_bar and (sw.cisd_dir == 1 ? close > sw.cisd_lvl : sw.cisd_dir == -1 ? close < sw.cisd_lvl : false)
                            if not consumed and not source_dead and not extreme_branched and not inv_vetoed and early_cross
                                cand = HtfCisdCandidate.new()
                                cand.ok := true
                                cand.pane := _pane
                                cand.dir := sw.cisd_dir
                                cand.htf_c2_idx := sw.cisd_htf_c2_idx
                                cand.c2_bar := sw.cisd_c2_bar
                                cand.c2_ext := sw.cisd_c2_ext
                                cand.cisd_lvl := sw.cisd_lvl
                                cand.cisd_inv_lvl := sw.cisd_inv_lvl
                                cand.cisd_set_high := sw.cisd_set_high
                                cand.cisd_set_low := sw.cisd_set_low
                                cand.cisd_set_wick_high := sw.cisd_set_wick_high
                                cand.cisd_set_wick_low := sw.cisd_set_wick_low
                                cand.cisd_anchor_bar := sw.cisd_anchor_bar
                                cand.anchor_t := sw.cisd_anchor_t
                                cand.src_x1 := sw.x1
                                cand.src_y := sw.y
                                if htf_cisd_candidate_better(cand, best)
                                    best := cand
    best
f_draw_on_chart_early_cisd(bool _en1, bool _en2, bool _en3) =>
    best = HtfCisdCandidate.new()
    if cisd_show and cisd_early_show and not barstate.isconfirmed
        best := htf_cisd_select_early_pane(htf_1_candles, 1, _en1, best)
        best := htf_cisd_select_early_pane(htf_2_candles, 2, _en2, best)
        best := htf_cisd_select_early_pane(htf_3_candles, 3, _en3, best)
        if best.ok
            early_x = math.max(best.anchor_t + 1, int(time))
            early_line = line.new(x1=best.anchor_t, y1=best.cisd_lvl, x2=early_x, y2=best.cisd_lvl, xloc=xloc.bar_time, color=best.dir == 1 ? cisd_bull_clr : cisd_bear_clr, style=cisd_early_line_style, width=cisd_early_line_width)
            bin.bin_ln.push(early_line)
    best

f_parse_sd_projection_levels(string _input) =>
    levels = array.new<float>()
    tokens = str.split(_input, ',')
    token_count = tokens.size()
    if token_count > 0
        for i = 0 to token_count - 1
            token = str.replace_all(str.replace_all(tokens.get(i), ' ', ''), '+', '')
            value = str.tonumber(token)
            if not na(value) and value > 0
                duplicate = false
                level_count = levels.size()
                if level_count > 0
                    for j = 0 to level_count - 1
                        if math.abs(levels.get(j) - value) <= 0.000001
                            duplicate := true
                            break
                if not duplicate
                    levels.push(value)
                    if levels.size() >= SD_PROJECTION_LEVEL_CAP
                        break
    levels

f_sd_source_better(SdProjectionSource _candidate, SdProjectionSource _current) =>
    _candidate.ok and (not _current.ok or nz(_candidate.recency_bar, -1) > nz(_current.recency_bar, -1))

    source

    source

        if ready3
            best := htf_cisd_select_pane(htf_3_candles, 3, _en3, best)
        if best.ok and not htf_cisd_event_consumed(best.dir, best.c2_bar, best.pane, best.src_x1, best.src_y)

            cisd_event_cleanup_state_cap()
            cisd_render_display_cap()

    spawned

method position_htf_sweeps(array<UDT_HTF_Candle> htf_candles) =>
    count = htf_candles.size()
    if count > 1
        for [i, candle] in htf_candles
            for [j, sweep] in candle.htf_sweeps
                sweep.x1 := candle.wick_x
                if i > 1
                    c3 = htf_candles.get(i - 2)
                    sweep.x2 := c3.wick_x
                else if i > 0
                    c2 = htf_candles.get(i - 1)
                    sweep.x2 := c2.candle_rigth
                else
                    sweep.x2 := candle.candle_rigth + 2
    htf_candles
method invalidate_sweep(UDT_Sweep sweep, UDT_HTF_Candle c2) =>
    not na(sweep.y) and (sweep.bull ? c2.c > sweep.y : c2.c < sweep.y)
method add_sweep_pair(UDT_HTF_Candle c1, UDT_HTF_Candle c2, bool bull) =>
    brk = bull ? c2.high_break_idx : c2.low_break_idx
    x1 = bull ? c1.h_idx : c1.l_idx
    y = bull ? c1.h : c1.l
    left = not na(brk) ? math.max(c2.o_idx, brk) : c2.o_idx
    htf_sweep = UDT_Sweep.new(x1=x1, x2=c2.c_idx, y=y, c2_left=left, c2_right=c2.c_idx, bull=bull, tf=c1.tf)
    ltf_sweep = UDT_Sweep.new(x1=x1, x2=c2.c_idx, y=y, c2_left=left, c2_right=c2.c_idx, bull=bull, tf=c1.tf)
    htf_sweep.cross_idx := brk
    htf_sweep.cross_triggered := true
    ltf_sweep.cross_idx := brk
    ltf_sweep.cross_triggered := true
    c1.htf_sweeps.push(htf_sweep)
    c1.ltf_sweeps.push(ltf_sweep)
    if bull
        c1.bull_sweep := true
    else
        c1.bear_sweep := true
    c1
method invalidate_sweeps(array<UDT_HTF_Candle> htf_candles, bool full_rescan) =>
    count = htf_candles.size()
    max_i = math.min(count - 1, HTF_SWEEP_STATE_CANDLES - 1)
    if count > 1 and max_i >= 1
        for i = max_i to 1 by 1
            c1 = htf_candles.get(i)
            c2_next = htf_candles.get(i - 1)
            for [j, sweep] in c1.ltf_sweeps
                htf_sweep = c1.htf_sweeps.get(j)
                if not sweep.removed and na(sweep.invalidated_on)
                    newest_closed_eval = sweep.last_closed_eval_o_idx
                    scan_start = full_rescan ? i - 1 : 0
                    for k = scan_start to 0 by 1
                        c2 = htf_candles.get(k)
                        skip_closed = c2.is_closed and not na(newest_closed_eval) and c2.o_idx <= newest_closed_eval
                        if not skip_closed
                            invalidated = sweep.invalidate_sweep(c2)
                            if sweep.x2 <= c2.c_idx and sweep.x2 > c2.o_idx
                                if not c2.is_closed
                                    sweep.invalidated := invalidated
                                    htf_sweep.invalidated := sweep.invalidated
                                else
                                    newest_closed_eval := c2.o_idx
                                    if invalidated and na(sweep.invalidated_on)
                                        sweep.invalidated_on := c2.o_idx
                                        htf_sweep.invalidated_on := sweep.invalidated_on
                                        break
                            else
                                if not c2.is_closed
                                    sweep.invalidated := invalidated
                                    htf_sweep.invalidated := sweep.invalidated
                                else
                                    newest_closed_eval := c2.o_idx
                                    if invalidated and na(sweep.invalidated_on)
                                        sweep.invalidated := invalidated
                                        sweep.invalidated_on := c2.o_idx
                                        htf_sweep.invalidated := sweep.invalidated
                                        htf_sweep.invalidated_on := sweep.invalidated_on
                                        break
                    sweep.last_closed_eval_o_idx := newest_closed_eval
                    htf_sweep.last_closed_eval_o_idx := newest_closed_eval
                if not sweep.formed and not sweep.removed and c2_next.is_closed
                    if sweep.invalidated and not na(sweep.invalidated_on)
                        sweep.removed := true
                        htf_sweep.removed := true
                    else
                        sweep.formed := true
                        htf_sweep.formed := true
        htf_candles
method detect_sweeps(array<UDT_HTF_Candle> htf_candles, bool full_rescan) =>
    count = htf_candles.size()
    if count > 1
        size = full_rescan ? math.min(4, count - 1) : 1
        for i = size to 1 by 1
            c1 = htf_candles.get(i)
            c2 = htf_candles.get(i - 1)
            if c2.h > c1.h and na(c2.high_break_idx)
                c2.high_break_idx := c2.h_idx
            if c2.l < c1.l and na(c2.low_break_idx)
                c2.low_break_idx := c2.l_idx
            if c1.htf_sweeps.size() < 2
                is_high_sweep = c2.h > c1.h and c2.c < c1.h
                is_low_sweep = c2.l < c1.l and c2.c > c1.l
                if is_high_sweep and not c1.bull_sweep
                    c1.add_sweep_pair(c2, true)
                if is_low_sweep and not c1.bear_sweep
                    c1.add_sweep_pair(c2, false)
            if i == 1
                for sweep in c1.ltf_sweeps
                    sweep.x2 := c2.c_idx
                    sweep.c2_left := not na(sweep.cross_idx) ? math.max(c2.o_idx, sweep.cross_idx) : c2.o_idx
                    sweep.c2_right := c2.c_idx
        htf_candles.invalidate_sweeps(full_rescan)
    htf_candles

f_htf_pane_cisd_core(array<UDT_HTF_Candle> candles, int c2_index, int dir) =>
    float level = na
    float inv_level = na
    int anchor_index = na
    if c2_index >= 0 and c2_index < candles.size()
        c2 = candles.get(c2_index)
        scan_upclose = dir == -1
        c2_upclose = c2.c > c2.o
        c2_downclose = c2.c < c2.o
        start_index = (scan_upclose ? c2_upclose : c2_downclose) ? c2_index : c2_index + 1
        last_scan_index = math.min(candles.size() - 1, start_index + HTF_PANE_CISD_SCAN_CANDLES - 1)
        scanned = 0
        started = false
        if start_index <= last_scan_index
            for scan_index = start_index to last_scan_index
                scan_candle = candles.get(scan_index)
                upclose = scan_candle.c > scan_candle.o
                downclose = scan_candle.c < scan_candle.o
                accepted = scan_upclose ? upclose : downclose
                opposite = scan_upclose ? downclose : upclose
                if accepted
                    started := true
                    scanned += 1
                    level := scan_candle.o
                    inv_level := QTU.cisdBodyInvAccum(inv_level, dir, scan_candle.o, scan_candle.c)
                    anchor_index := scan_index
                    if scanned >= HTF_PANE_CISD_SCAN_CANDLES
                        break
                else if started and opposite
                    break
    [level, inv_level, anchor_index]
f_htf_pane_cisd_inv_level_veto(array<UDT_HTF_Candle> candles, int dir, float inv_level, int anchor_index) =>
    violated = false
    if dir != 0 and not na(inv_level) and not na(anchor_index) and anchor_index > 0
        // Pane arrays are newest-first. Start at the HTF candle immediately after the CISD anchor.
        for scan_index = anchor_index - 1 to 0
            scan_candle = candles.get(scan_index)
            if scan_candle.is_closed
                violated := dir == 1 ? scan_candle.c < inv_level : dir == -1 ? scan_candle.c > inv_level : false
                if violated
                    break
    violated

f_htf_pane_cisd_confirmation(array<UDT_HTF_Candle> candles, int c2_index, int dir, float level, float c2_ext, int source_invalidated_on) =>
    int confirm_index = na
    bool extreme_branched = false
    if not na(level) and not na(c2_ext)
        for scan_index = c2_index to 0
            scan_candle = candles.get(scan_index)
            if scan_index < c2_index and scan_candle.is_closed
                extreme_branched := extreme_branched or QTU.cisdC2Invalidates(dir, c2_ext, scan_candle.l, scan_candle.h)
            source_dead = not na(source_invalidated_on) and source_invalidated_on <= scan_candle.c_idx
            crossed = scan_candle.is_closed and (dir == 1 ? scan_candle.c > level : scan_candle.c < level)
            if source_dead or extreme_branched
                break
            if crossed
                confirm_index := scan_index
                break
    confirm_index
f_htf_pane_cisd_preconfirm_valid(array<UDT_HTF_Candle> candles, int c2_index, int dir, float c2_ext, int source_invalidated_on) =>
    valid = not na(c2_ext)
    bool extreme_branched = false
    if valid
        for scan_index = c2_index to 0
            scan_candle = candles.get(scan_index)
            if scan_index < c2_index and scan_candle.is_closed
                extreme_branched := extreme_branched or QTU.cisdC2Invalidates(dir, c2_ext, scan_candle.l, scan_candle.h)
            source_dead = not na(source_invalidated_on) and source_invalidated_on <= scan_candle.c_idx
            if source_dead or extreme_branched
                valid := false
                break
    valid
f_htf_pane_cisd_post_confirm_state(array<UDT_HTF_Candle> candles, int confirm_index, int dir, float inv_level) =>
    invalid = false
    pending = false
    if not na(inv_level) and confirm_index > 0
        for scan_index = confirm_index - 1 to 0
            scan_candle = candles.get(scan_index)
            if scan_candle.is_closed
                invalid := dir == 1 ? scan_candle.c < inv_level : dir == -1 ? scan_candle.c > inv_level : false
                if invalid
                    break
            else
                pending := pending or (dir == 1 ? scan_candle.l < inv_level : dir == -1 ? scan_candle.h > inv_level : false)
    [invalid, pending]
f_draw_htf_pane_cisds(array<UDT_HTF_Candle> candles, bool enabled) =>
    candle_count = candles.size()
    if enabled and htf_cisd_show and candle_count > 1
        for c1_index = 1 to candle_count - 1
            c1 = candles.get(c1_index)
            c2_index = c1_index - 1
            c2 = candles.get(c2_index)
            if c2.is_closed and c1.htf_sweeps.size() > 0
                drawn = false
                for sweep in c1.htf_sweeps
                    if not drawn
                        dir = sweep.bull ? -1 : 1
                        c2_ext = dir == 1 ? c2.l : c2.h
                        [level, inv_level, anchor_index] = f_htf_pane_cisd_core(candles, c2_index, dir)
                        inv_vetoed = f_htf_pane_cisd_inv_level_veto(candles, dir, inv_level, anchor_index)
                        confirm_index = f_htf_pane_cisd_confirmation(candles, c2_index, dir, level, c2_ext, sweep.invalidated_on)
                        [invalid, pending] = f_htf_pane_cisd_post_confirm_state(candles, confirm_index, dir, inv_level)
                        preconfirm_valid = f_htf_pane_cisd_preconfirm_valid(candles, c2_index, dir, c2_ext, sweep.invalidated_on)
                        if not inv_vetoed and not invalid and not pending and not na(confirm_index) and not na(level) and not na(inv_level) and not na(anchor_index)
                            anchor_x = candles.get(anchor_index).candle_left
                            crossover_x = candles.get(confirm_index).candle_rigth
                            pane_line = line.new(x1=anchor_x, y1=level, x2=math.max(anchor_x + 1, crossover_x + HTF_PANE_CISD_EXTEND_BARS), y2=level, xloc=xloc.bar_index, color=dir == 1 ? htf_cisd_bull_clr : htf_cisd_bear_clr, style=line.style_solid, width=htf_cisd_line_width)
                            bin.bin_ln.push(pane_line)
                            drawn := true
                        else if not inv_vetoed and htf_early_cisd_show and na(confirm_index) and preconfirm_valid and not na(level) and not na(inv_level) and not na(anchor_index)
                            live_candle = candles.get(0)
                            early_cross = not live_candle.is_closed and (dir == 1 ? live_candle.c > level : live_candle.c < level)
                            if early_cross
                                anchor_x = candles.get(anchor_index).candle_left
                                early_x = live_candle.candle_rigth
                                early_line = line.new(x1=anchor_x, y1=level, x2=math.max(anchor_x + 1, early_x + HTF_PANE_CISD_EXTEND_BARS), y2=level, xloc=xloc.bar_index, color=dir == 1 ? htf_cisd_bull_clr : htf_cisd_bear_clr, style=htf_early_cisd_line_style, width=htf_early_cisd_line_width)
                                bin.bin_ln.push(early_line)
                                drawn := true
    0

f_sd_source_from_htf_candle(UDT_HTF_Candle _candle, int _dir, int _recency_bar) =>
    source = SdProjectionSource.new()
    if _dir != 0 and not na(_candle.o_idx) and not na(_candle.o) and not na(_candle.c) and not na(_candle.h) and not na(_candle.l)
        source.ok := true
        source.dir := _dir
        source.recency_bar := _recency_bar
        source.anchor_bar := _candle.o_idx
        source.body_high := math.max(_candle.o, _candle.c)
        source.body_low := math.min(_candle.o, _candle.c)
        source.wick_high := _candle.h
        source.wick_low := _candle.l
    source

f_select_htf_pane_cisd_sd(array<UDT_HTF_Candle> _candles, bool _enabled, SdProjectionSource _best) =>
    best = _best
    candle_count = _candles.size()
    if _enabled and htf_cisd_show and candle_count > 1
        for c1_index = 1 to candle_count - 1
            c1 = _candles.get(c1_index)
            c2_index = c1_index - 1
            c2 = _candles.get(c2_index)
            if c2.is_closed and c1.htf_sweeps.size() > 0
                selected = false
                for sweep in c1.htf_sweeps
                    if not selected
                        dir = sweep.bull ? -1 : 1
                        c2_ext = dir == 1 ? c2.l : c2.h
                        [level, inv_level, anchor_index] = f_htf_pane_cisd_core(_candles, c2_index, dir)
                        inv_vetoed = f_htf_pane_cisd_inv_level_veto(_candles, dir, inv_level, anchor_index)
                        confirm_index = f_htf_pane_cisd_confirmation(_candles, c2_index, dir, level, c2_ext, sweep.invalidated_on)
                        [invalid, pending] = f_htf_pane_cisd_post_confirm_state(_candles, confirm_index, dir, inv_level)
                        preconfirm_valid = f_htf_pane_cisd_preconfirm_valid(_candles, c2_index, dir, c2_ext, sweep.invalidated_on)
                        confirmed_visible = not inv_vetoed and not invalid and not pending and not na(confirm_index) and not na(level) and not na(inv_level) and not na(anchor_index)
                        early_visible = false
                        if not inv_vetoed and htf_early_cisd_show and na(confirm_index) and preconfirm_valid and not na(level) and not na(inv_level) and not na(anchor_index)
                            live_candle = _candles.get(0)
                            early_visible := not live_candle.is_closed and (dir == 1 ? live_candle.c > level : live_candle.c < level)
                        if confirmed_visible or early_visible
                            anchor_candle = _candles.get(anchor_index)
                            candidate = f_sd_source_from_htf_candle(anchor_candle, dir, c2.o_idx)
                            if f_sd_source_better(candidate, best)
                                best := candidate
                            selected := true
    best

f_select_htf_c2_sd(array<UDT_HTF_Candle> _candles, bool _enabled, SdProjectionSource _best) =>
    best = _best
    candle_count = _candles.size()
    if _enabled and candle_count > 1
        for c1_index = 1 to candle_count - 1
            c1 = _candles.get(c1_index)
            c2 = _candles.get(c1_index - 1)
            if c2.is_closed and c1.htf_sweeps.size() > 0
                selected = false
                for sweep in c1.htf_sweeps
                    if not selected
                        c2_inside_c1 = not sweep.invalidate_sweep(c2)
                        valid_c2 = not sweep.invalidated and not sweep.removed and c2_inside_c1
                        if valid_c2
                            dir = sweep.bull ? -1 : 1
                            candidate = f_sd_source_from_htf_candle(c2, dir, c2.o_idx)
                            if f_sd_source_better(candidate, best)
                                best := candidate
                            selected := true
    best

f_draw_sd_projections(HtfCisdCandidate _early, bool _pane_cisd_1, bool _pane_cisd_2, bool _pane_cisd_3, bool _pane_1, bool _pane_2, bool _pane_3) =>
    if sd_projection_show
        source = SdProjectionSource.new()
        if sd_projection_source_input == 'On-chart CISD'
            source := f_latest_displaying_on_chart_sd()
            early_source = f_sd_source_from_candidate(_early)
            if f_sd_source_better(early_source, source) or early_source.ok and source.ok and nz(early_source.recency_bar, -1) == nz(source.recency_bar, -1)
                source := early_source
        else if sd_projection_source_input == 'HTF Pane CISD'
            source := f_select_htf_pane_cisd_sd(htf_1_candles, _pane_cisd_1, source)
            source := f_select_htf_pane_cisd_sd(htf_2_candles, _pane_cisd_2, source)
            source := f_select_htf_pane_cisd_sd(htf_3_candles, _pane_cisd_3, source)
        else
            source := f_select_htf_c2_sd(htf_1_candles, _pane_1, source)
            source := f_select_htf_c2_sd(htf_2_candles, _pane_2, source)
            source := f_select_htf_c2_sd(htf_3_candles, _pane_3, source)
        range_high = sd_anchor_mode_input == 'High to Low' ? source.wick_high : source.body_high
        range_low = sd_anchor_mode_input == 'High to Low' ? source.wick_low : source.body_low
        price_range = range_high - range_low
        if source.ok and not na(source.anchor_bar) and not na(price_range) and price_range > syminfo.mintick * 0.5
            levels = f_parse_sd_projection_levels(sd_projection_levels_input)
            level_count = levels.size()
            if level_count > 0
                projection_x1 = source.anchor_bar
                projection_x2 = projection_x1 + SD_PROJECTION_EXTEND_BARS

f_dealing_range_bounds(array<UDT_HTF_Candle> _candles) =>
    float range_top = na
    float range_bottom = na
    int pane_left = na
    int pane_right = na
    int high_anchor = na
    int low_anchor = na
    for candle in _candles
        positioned = not na(candle.candle_left) and not na(candle.candle_rigth)
        if positioned and not na(candle.h) and not na(candle.l)
            if na(range_top) or candle.h > range_top
                range_top := candle.h
                high_anchor := candle.h_idx
            if na(range_bottom) or candle.l < range_bottom
                range_bottom := candle.l
                low_anchor := candle.l_idx
            pane_left := na(pane_left) ? candle.candle_left : math.min(pane_left, candle.candle_left)
            pane_right := na(pane_right) ? candle.candle_rigth : math.max(pane_right, candle.candle_rigth)
    [range_top, range_bottom, pane_left, pane_right, high_anchor, low_anchor]

method draw_dealing_range(DealingRangeVisuals this, array<UDT_HTF_Candle> _candles, bool _enabled, string _mode, string _display, int _offset) =>
    [range_top, range_bottom, pane_left, pane_right, high_anchor, low_anchor] = f_dealing_range_bounds(_candles)
    on_chart = _display == 'On Chart Overlay'
    valid_anchors = not on_chart or (not na(high_anchor) and not na(low_anchor))
    valid_range = _enabled and valid_anchors and not na(range_top) and not na(range_bottom) and not na(pane_left) and not na(pane_right) and range_top - range_bottom > syminfo.mintick * 0.5
    if valid_range
        pane_width = math.max(1, pane_right - pane_left)
        int draw_left = pane_left
        int draw_right = pane_right
        if on_chart
            draw_left := math.min(high_anchor, low_anchor)
            draw_right := math.max(high_anchor, low_anchor)
            if draw_right == draw_left
                draw_right += 1
        else if _display == 'HTF Pane Offset'
            draw_left := bar_index + _offset
            draw_right := draw_left + pane_width
        range_size = range_top - range_bottom
        equilibrium = range_bottom + range_size * 0.5
        quadrant_mode = _mode == 'Quadrants'
        ratios = array.from(1.0, 0.75, 0.5, 0.25, 0.0)
        names = array.from('1', '0.75', '0.5', '0.25', '0')
        for i = 0 to 4
            level_active = quadrant_mode or i == 0 or i == 2 or i == 4
            ratio = ratios.get(i)
            price = range_bottom + range_size * ratio
            level_color = ratio >= 0.5 ? dealing_range_premium_color : dealing_range_discount_color
            level_style = line.style_dotted
            this.update_dealing_range_level(i, level_active, draw_left, draw_right, price, level_color, level_style, names.get(i))
        this.premium_box := f_update_dealing_range_box(this.premium_box, draw_left, draw_right, range_top, equilibrium, dealing_range_premium_color)
        this.discount_box := f_update_dealing_range_box(this.discount_box, draw_left, draw_right, equilibrium, range_bottom, dealing_range_discount_color)
    else
        this.clear_dealing_range()
    this

method detect_htf_candle(array<UDT_HTF_Candle> htf_candles, bool enabled, string tf) =>
    UDT_HTF_Candle htf_candle = na
    tfTime = time(tf)
    regular_begins = ta.change(tfTime) != 0
    custom_daily = htf_custom_daily_open and tf_sec <= TF_SEC_60 and QTU.isDailyTf(tf)
    tf_session_begins = custom_daily ? htf_custom_daily_begin : regular_begins
    if enabled
        if tf_session_begins or htf_candles.size() == 0
            candle = UDT_HTF_Candle.new(tf=tf, htf_sweeps=array.new<UDT_Sweep>(), ltf_sweeps=array.new<UDT_Sweep>())
            candle.o := open
            candle.c := close
            candle.h := high
            candle.l := low
            candle.o_idx := bar_index
            candle.c_idx := bar_index
            candle.h_idx := bar_index
            candle.l_idx := bar_index
            candle.ot := time
            candle.bull := QTU.isBullishCandle(candle.c, candle.o, candle.h, candle.l)
            if htf_candles.size() > 0
                last_candle = htf_candles.get(0)
                last_candle.is_closed := true
                last_candle.ct := time
                last_candle.ct
            htf_candle := candle
            htf_candle
        else if htf_candles.size() > 0
            candle = htf_candles.first()
            candle.c := close
            candle.c_idx := bar_index
            candle.ct := time
            if high > candle.h
                candle.h := high
                candle.h_idx := bar_index
                candle.h_idx
            if low < candle.l
                candle.l := low
                candle.l_idx := bar_index
                candle.l_idx
            candle.bull := QTU.isBullishCandle(candle.c, candle.o, candle.h, candle.l)
            candle.bull
    htf_candle

method draw_htf_label(array<UDT_HTF_Candle> htf_candles, string tf, float shared_top, float shared_bottom, int close_t, color label_clr, bool src_lbl) =>
    float y_top = na
    float y_bottom = na
    int x_min = na
    int x_max = na
    int cd_close = na
    for [index, candle] in htf_candles
        y_top := na(y_top) ? candle.h : math.max(y_top, candle.h)
        y_bottom := na(y_bottom) ? candle.l : math.min(y_bottom, candle.l)
        x_min := na(x_min) ? candle.wick_x : math.min(x_min, candle.wick_x)
        x_max := na(x_max) ? candle.wick_x : math.max(x_max, candle.wick_x)
        cd_close := index == 0 and htf_custom_daily_open and QTU.isDailyTf(tf) ? candle.ot + 86400000 : cd_close
    x = not na(x_min) and not na(x_max) ? math.round(math.avg(x_min, x_max)) : na
    top_y = htf_label_alignment == 'Align' and not na(shared_top) ? shared_top : y_top
    bottom_y = htf_label_alignment == 'Align' and not na(shared_bottom) ? shared_bottom : y_bottom
    pad = QTU.labelPad(top_y, bottom_y, syminfo.mintick)
    top_y += pad
    bottom_y -= pad
    txt = QTU.htfSourceLabel(tf, src_lbl)
    rem = QTU.remTime(na(cd_close) ? close_t : cd_close, timenow)
    has_rem = str.length(rem) > 0
    top_on = htf_label_position == 'Both' or htf_label_position == 'Top'
    bottom_on = htf_label_position == 'Both' or htf_label_position == 'Bottom'
    if top_on and not na(top_y) and not na(x)
        if htf_label_show
            f_push_htf_lbl(x, top_y, txt + (htf_timer_show and has_rem ? "\n" : ""), label.style_label_down, label_clr, size.small)
        if htf_timer_show and has_rem
            f_push_htf_lbl(x, top_y, rem, label.style_label_down, htf_timer_color, size.small)
    if bottom_on and not na(bottom_y) and not na(x)
        if htf_label_show
            f_push_htf_lbl(x, bottom_y, (htf_timer_show and has_rem ? "\n" : "") + txt, label.style_label_up, label_clr, size.small)
        if htf_timer_show and has_rem
            f_push_htf_lbl(x, bottom_y, rem, label.style_label_up, htf_timer_color, size.small)
    htf_candles
f_draw_sweep_set(array<UDT_HTF_Candle> htf_candles, bool ltf, HtfPaneCfg cfg) =>
    if cfg.sweep_cap == 1
        int newest_candle_index = na
        int newest_sweep_index = na
        int newest_event_index = na
        for [i, candle] in htf_candles
            sweep_count = ltf ? array.size(candle.ltf_sweeps) : array.size(candle.htf_sweeps)
            if sweep_count > 0
                for j = 0 to sweep_count - 1
                    UDT_Sweep sweep = ltf ? array.get(candle.ltf_sweeps, j) : array.get(candle.htf_sweeps, j)
                    c2_inside_c1 = false
                    if i > 0
                        UDT_HTF_Candle c2_ref = array.get(htf_candles, i - 1)
                        c2_inside_c1 := not sweep.invalidate_sweep(c2_ref)
                    valid_c1_sweep = not sweep.invalidated and not sweep.removed and c2_inside_c1
                    if valid_c1_sweep
                        event_index = nz(sweep.cross_idx, nz(sweep.c2_right, sweep.x2))
                        if na(newest_event_index) or event_index > newest_event_index
                            newest_event_index := event_index
                            newest_candle_index := i
                            newest_sweep_index := j
        if not na(newest_candle_index) and not na(newest_sweep_index)
            newest_candle = htf_candles.get(newest_candle_index)
            UDT_Sweep newest_sweep = ltf ? newest_candle.ltf_sweeps.get(newest_sweep_index) : newest_candle.htf_sweeps.get(newest_sweep_index)
            int c3_open_idx = newest_candle_index >= 2 ? htf_candles.get(newest_candle_index - 2).o_idx : na
            newest_sweep.draw_sweep(ltf, cfg, c3_open_idx)
    else
        counted = 0
        for [i, candle] in htf_candles
            sweep_count = ltf ? array.size(candle.ltf_sweeps) : array.size(candle.htf_sweeps)
            if sweep_count > 0
                c2_closed = ltf or (i > 0 and array.get(htf_candles, i - 1).is_closed)
                for j = 0 to sweep_count - 1
                    UDT_Sweep sweep = ltf ? array.get(candle.ltf_sweeps, j) : array.get(candle.htf_sweeps, j)
                    c2_live_valid = true
                    if i > 0
                        UDT_HTF_Candle c2_ref = array.get(htf_candles, i - 1)
                        c2_live_valid := not sweep.invalidate_sweep(c2_ref)
                    if not sweep.invalidated and not sweep.removed and c2_live_valid
                        within_cap = not c2_closed or counted < cfg.sweep_cap
                        if within_cap
                            int c3_open_idx = i >= 2 ? htf_candles.get(i - 2).o_idx : na
                            sweep.draw_sweep(ltf, cfg, c3_open_idx)
                        if c2_closed and within_cap
                            counted += 1
                            if counted >= cfg.sweep_cap
                                break
            if counted >= cfg.sweep_cap
                break
    0
