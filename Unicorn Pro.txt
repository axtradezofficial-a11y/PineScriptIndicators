method Reorder(CandleSet candleSet, HTF_Settings sett, int base_offset) =>
    for l in candleSet.sweeps
        line.delete(l)
    array.clear(candleSet.sweeps)
    int size = array.size(candleSet.candles)
    if size > 0
        float max_high = na
        int center_x = bar_index + base_offset + math.round((((sett.width + sett.buffer) * size) - sett.buffer) / 2)
        for i = size - 1 to 0
            Candle candle = array.get(candleSet.candles, i)
            max_high := na(max_high) ? candle.h : math.max(max_high, candle.h)
            int t_buffer = base_offset + ((sett.width + sett.buffer) * (size - i - 1))
            int posX = bar_index + t_buffer
            int midX = posX + (sett.width / 2)
            box.set_left(candle.body, posX)
            box.set_right(candle.body, posX + sett.width)
            line.set_x1(candle.wick_up, midX)
            line.set_x2(candle.wick_up, midX)
            line.set_x1(candle.wick_down, midX)
            line.set_x2(candle.wick_down, midX)
            if i < size - 1
                Candle prev = array.get(candleSet.candles, i + 1)
                int prevPosX = base_offset + ((sett.width + sett.buffer) * (size - (i + 1) - 1))
                int prevMidX = bar_index + prevPosX + (sett.width / 2)
                if sett.show_sweeps
                    if candle.h > prev.h and candle.c < prev.h
                        array.push(candleSet.sweeps, line.new(prevMidX, prev.h, posX + sett.width, prev.h, color=sett.line_color))
                    if candle.l < prev.l and candle.c > prev.l
                        array.push(candleSet.sweeps, line.new(prevMidX, prev.l, posX + sett.width, prev.l, color=sett.line_color))
        string tf_str = f_tfLabel(candleSet.settings.htf)
        string parsed_size = sett.lbl_size == "tiny" ? size.tiny : sett.lbl_size == "small" ? size.small : sett.lbl_size == "large" ? size.large : sett.lbl_size == "huge" ? size.huge : size.normal
        if na(candleSet.tf_label)
            candleSet.tf_label := label.new(center_x, max_high, tf_str, color=color.new(color.black, 100), style=label.style_label_down, yloc=yloc.price)
        label.set_xy(candleSet.tf_label, center_x, max_high)
        label.set_text(candleSet.tf_label, tf_str)
        label.set_textcolor(candleSet.tf_label, sett.lbl_color)
        label.set_size(candleSet.tf_label, parsed_size)

method Monitor(CandleSet candleSet, HTF_Settings sett) =>
    bool isNew = ta.change(time(candleSet.settings.htf)) != 0
    if isNew 
        Candle candle = Candle.new(open, close, high, low, box(na), line(na), line(na))
        bool bull = candle.c > candle.o
        candle.body := box.new(bar_index, math.max(candle.o, candle.c), bar_index + 2, math.min(candle.o, candle.c), border_color = bull ? sett.bull_border : sett.bear_border, border_width = 1, bgcolor = bull ? sett.bull_body : sett.bear_body)
        candle.wick_up := line.new(bar_index, candle.h, bar_index, math.max(candle.o, candle.c), color=bull ? sett.bull_wick : sett.bear_wick)
        candle.wick_down := line.new(bar_index, math.min(candle.o, candle.c), bar_index, candle.l, color=bull ? sett.bull_wick : sett.bear_wick)
        array.unshift(candleSet.candles, candle)
        if array.size(candleSet.candles) > candleSet.settings.max_display
            Candle del = array.pop(candleSet.candles)
            box.delete(del.body)
            line.delete(del.wick_up)
            line.delete(del.wick_down)
    candleSet

method Update(CandleSet candleSet, HTF_Settings sett, int base_offset) =>
    if array.size(candleSet.candles) > 0
        Candle candle = array.get(candleSet.candles, 0)
        candle.h := math.max(high, candle.h)
        candle.l := math.min(low, candle.l)
        candle.c := close
        bool bull = candle.c > candle.o
        box.set_top(candle.body, math.max(candle.o, candle.c))
        box.set_bottom(candle.body, math.min(candle.o, candle.c))
        box.set_bgcolor(candle.body, bull ? sett.bull_body : sett.bear_body)
        box.set_border_color(candle.body, bull ? sett.bull_border : sett.bear_border)
        line.set_y1(candle.wick_up, candle.h)
        line.set_y2(candle.wick_up, math.max(candle.o, candle.c))
        line.set_y1(candle.wick_down, candle.l)
        line.set_y2(candle.wick_down, math.min(candle.o, candle.c))
        if barstate.islast
            candleSet.Reorder(sett, base_offset)
    candleSet

f_parseExtras(str) =>
    parts = str.split(str, ",")
    arr   = array.new_float(0)
    for p in parts
        v = str.tonumber(str.trim(p))
        if not na(v)
            array.push(arr, v)
    arr

f_findLevelOrigin(isBull, lvl) =>
    int offset = 0
    for j = 1 to 400
        match = isBull ? (math.abs(low[j] - lvl) < syminfo.mintick) : (math.abs(high[j] - lvl) < syminfo.mintick)
        if match
            offset := j
            break
    offset

f_findOHLCOrigin(isBull, lvl, tf) =>
    htf_curr_start = ta.valuewhen(ta.change(time(tf)), bar_index, 0)
    htf_prev_start = ta.valuewhen(ta.change(time(tf)), bar_index, 1)
    htf_prev_end   = htf_curr_start - 1
    offset = 0
    if not na(htf_prev_start) and not na(htf_prev_end)
        start_j = bar_index - htf_prev_end
        end_j   = bar_index - htf_prev_start
        for j = math.max(1, start_j) to math.min(490, end_j)
            match = isBull ? (math.abs(low[j] - lvl) < syminfo.mintick) : (math.abs(high[j] - lvl) < syminfo.mintick)
            if match
                offset := j
                break
    offset

f_alreadySweptBeforeOrigin(isLow, lvl, offset) =>
    bool swept = false
    if not na(offset) and offset > 1
        for i = 1 to offset - 1
            if isLow ? (low[i] <= lvl) : (high[i] >= lvl)
                swept := true
                break
    swept

f_alreadyTouchedPair(isLowLevel, lvl1, lvl2, tf) =>
    int barsSinceStart = ta.barssince(ta.change(time(tf)))
    bool touched1 = false
    bool touched2 = false
    if not na(barsSinceStart) and barsSinceStart > 0
        int limit = math.min(barsSinceStart, 300)
        for i = 1 to limit
            if isLowLevel
                if not touched1 and low[i] <= lvl1
                    touched1 := true
                if not touched2 and low[i] <= lvl2
                    touched2 := true
            else
                if not touched1 and high[i] >= lvl1
                    touched1 := true
                if not touched2 and high[i] >= lvl2
                    touched2 := true
            if touched1 and touched2
                break
    [touched1, touched2]

var float[] _bkSizeHistory = array.new_float(0)

f_sizeOk(bkSize) =>
    pip = syminfo.mintick * (syminfo.type == "forex" ? 10.0 : 1.0)
    sz  = i_customSizeMode == "Pips" ? bkSize / pip : bkSize
    if i_sizeFilter == "All"
        true
    else if i_sizeFilter == "Custom"
        minOk = i_customMin == 0.0 or sz >= i_customMin
        maxOk = i_customMax == 0.0 or sz <= i_customMax
        minOk and maxOk
    else
        array.push(_bkSizeHistory, sz)
        if array.size(_bkSizeHistory) > 100
            array.shift(_bkSizeHistory)
        n = array.size(_bkSizeHistory)
        if n < 3
            true
        else
            sorted = array.copy(_bkSizeHistory)
            array.sort(sorted)

f_findBB(isBull, pivotOffset) =>
    bbTop = float(na)
    bbBot = float(na)
    bbLen = 0
    baseIdx = na(pivotOffset) ? 1 : pivotOffset
    startIdx = baseIdx
    if not na(pivotOffset)
        for k = baseIdx - 1 to 1
            if isBull ? (close[k] > open[k]) : (close[k] < open[k])
                startIdx := k
            else
                break
    for i = startIdx to startIdx + 8
        if isBull ? (close[i] > open[i]) : (close[i] < open[i])
            bbTop := na(bbTop) ? high[i] : math.max(bbTop, high[i])
            bbBot := na(bbBot) ? low[i]  : math.min(bbBot, low[i])
            bbLen := (i - startIdx) + 1
        else
            if not na(bbTop)
                break
    if na(bbTop) and not na(pivotOffset)
        bbTop := high[pivotOffset]
        bbBot := low[pivotOffset]
        bbLen := 1
    [bbTop, bbBot, bbLen]

                    array.push(idxs, i)
    [array.size(tops) > 0, tops, bots, idxs]

f_registerSetup(isBull, bbTop, bbBot, sweepLevel, sweepBar, sweepOffset, bbLeft, manipExtreme, hasFvg, fvgTops, fvgBots, fvgLefts, fvgRights, sweepLbl) =>

    if showThis
        entry    = isBull ? bbTop : bbBot
        invalLvl = i_swingInval ? manipExtreme : (isBull ? bbBot : bbTop)
        riskSz   = math.abs(entry - invalLvl)
        if f_sizeOk(riskSz) and f_biasOk(isBull)
            extras  = f_parseExtras(i_extraStr)
            nExtras = array.size(extras)
            projEntry = i_projCalc == "Confirmation Close" ? close : (isBull ? bbTop : bbBot)
            risk = math.abs(projEntry - invalLvl)
            mainTgt = isBull ? invalLvl + (i_rrMain + 1.0) * risk : invalLvl - (i_rrMain + 1.0) * risk
            extraPrices = array.new_float(0)

                mult = array.get(extras, ei)
                t    = isBull ? invalLvl + (mult + 1.0) * risk : invalLvl - (mult + 1.0) * risk
                array.push(extraPrices, t)
            extRight = bar_index + 20
            zCol = isBull ? i_bullColor : i_bearColor
            lblTxt = isUni ? (isBull ? "+Unicorn" : "-Unicorn")
                           : (isBull ? "+Breaker" : "-Breaker")
            zBox = box.new(bbLeft, bbTop, extRight, bbBot,
                           border_color=color.new(zCol, 100),
                           border_width=0,
                           bgcolor=zCol,
                           extend=extend.none,
                           text=lblTxt,
                           text_size=f_lblSize(i_lblSize),
                           text_color=color.black,
                           text_halign=text.align_right,
                           text_valign=text.align_center)
            fBoxArr = array.new<box>(0)
            ceLines = array.new<line>(0)
            if hasFvg
                fvgCol = isBull ? i_bullFvgColor : i_bearFvgColor
                for i = 0 to array.size(fvgTops) - 1
                    ft = array.get(fvgTops, i)

f_updateSetups(prevLabel) =>
    string retLabel = prevLabel
    n = array.size(setups)
    if n > 0
        for i = n - 1 to 0
            s = array.get(setups, i)
            if s.state == 4 and i_discardInval
                continue
            if s.state >= 3
                continue
            isBull = s.isBull
            if s.state == 1
                retested = isBull ? (low <= s.zoneTop and low >= s.zoneBot) : (high >= s.zoneBot and high <= s.zoneTop)
                if retested
                    s.state := 2
                    newCol = isBull ? i_bullRetColor : i_bearRetColor
                    box.set_bgcolor(s.zoneBox, newCol)
                    box.set_border_color(s.zoneBox, color.new(newCol, 100))
                    box.set_text_color(s.zoneBox, color.black)
            if s.state == 1 or s.state == 2
                box.set_right(s.zoneBox,  bar_index + 20)
                if not na(s.bb50Line)
                    extendLine = s.isBull ? i_extendBB50Bull : i_extendBB50Bear
                    if extendLine
                        boxLeft = box.get_left(s.zoneBox)
                        line.set_x2(s.bb50Line, math.max(boxLeft, bar_index + 20 - 5))
                if array.size(s.fvgBoxes) > 0
                    for fi = 0 to array.size(s.fvgBoxes) - 1
                        curFvgBox = array.get(s.fvgBoxes, fi)
                        if (s.isBull and i_extendFVGBull) or (s.isBull == false and i_extendFVGBear)
                            box.set_right(curFvgBox, bar_index + 20)
                            if array.size(s.fvgCeLines) > fi
                                line.set_x2(array.get(s.fvgCeLines, fi), bar_index + 20)
            if i_modelMode == "Breaker + Unicorn" and not s.isUnicorn
                bool fFound = false
                float[] fTops = na
                float[] fBots = na
                int[] fIdxs = na
                [found, tops, bots, idxs] = f_findFvg(s.isBull, s.zoneTop, s.zoneBot, s.sweepBar)
                fFound := found
                fTops  := tops
                fBots  := bots
                fIdxs  := idxs
                if fFound
                    s.isUnicorn := true
                    box.set_text(s.zoneBox, s.isBull ? "+Unicorn" : "-Unicorn")
                    fvgCol = s.isBull ? i_bullFvgColor : i_bearFvgColor
                    for i = 0 to array.size(fTops) - 1
                        ft = array.get(fTops, i)
                        fb = array.get(fBots, i)
                        fBox = box.new(bar_index - array.get(fIdxs, i) - 2, ft, bar_index - array.get(fIdxs, i), fb,
                                         border_color=color.new(fvgCol, 100), border_width=0, bgcolor=fvgCol)
                        array.push(s.fvgBoxes, fBox)
                        fMid = (ft + fb) / 2.0
                        if (s.isBull and i_showCEBull) or (not s.isBull and i_showCEBear)
                            cel = line.new(bar_index - array.get(fIdxs, i) - 2, fMid, bar_index - array.get(fIdxs, i), fMid, color=fvgCol, width=1, style=line.style_dotted)
                            array.push(s.fvgCeLines, cel)
            if s.state == 1 or s.state == 2
                hitTarget = isBull ? high >= s.mainTarget : low <= s.mainTarget
                if hitTarget
                    s.state := 3
                    box.set_right(s.zoneBox,  bar_index)
                    if not na(s.bb50Line)
                        extendLine = s.isBull ? i_extendBB50Bull : i_extendBB50Bear
                        if extendLine
                            boxLeft = box.get_left(s.zoneBox)
                            line.set_x2(s.bb50Line, math.max(boxLeft, bar_index - 5))
                    if array.size(s.fvgBoxes) > 0
                        for fi = 0 to array.size(s.fvgBoxes) - 1
                            curFvgBox = array.get(s.fvgBoxes, fi)
                            extendFVG = s.isBull ? i_extendFVGBull : i_extendFVGBear
                            if extendFVG
                                box.set_right(curFvgBox, bar_index)
                                if array.size(s.fvgCeLines) > fi
                                    line.set_x2(array.get(s.fvgCeLines, fi), bar_index)
                    if i_alertEnable and i_alertTarget and not s.alertedTarget
                        alert("Unicorn Model° Target Reached | " + syminfo.ticker + " | " + (isBull ? "BULL" : "BEAR") + " | " + str.tostring(s.mainTarget, format.mintick), alert.freq_once_per_bar)
                        s.alertedTarget := true
            if s.state == 1 or s.state == 2
                invalHit   = isBull ? (low <= s.invalidLevel) : (high >= s.invalidLevel)
                invalWarn  = not invalHit and (isBull ? (low < s.invalidLevel + s.riskSize * 0.15) : (high > s.invalidLevel - s.riskSize * 0.15))
                if invalWarn and not s.alertedInvalWarn
                    if i_alertEnable and i_alertWarn
                        alert("Unicorn Model° Invalidation Warning | " + syminfo.ticker + " | " + (isBull ? "BULL" : "BEAR") + " | Level: " + str.tostring(s.invalidLevel, format.mintick), alert.freq_once_per_bar)
                    s.alertedInvalWarn := true
                if invalHit
                    s.state := 4
                    box.set_bgcolor(s.zoneBox,       i_invalColor)
                    box.set_border_color(s.zoneBox,  color.new(#9e9e9e, 100))
                    box.set_right(s.zoneBox,         bar_index)
                    box.set_text_color(s.zoneBox,    color.new(#9e9e9e, 40))
                    box.set_text(s.zoneBox,          s.isUnicorn ? (isBull ? "✕ +Unicorn" : "✕ -Unicorn") : (isBull ? "✕ +Breaker" : "✕ -Breaker"))
                    if not na(s.bb50Line)
                        extendLine = s.isBull ? i_extendBB50Bull : i_extendBB50Bear
                        if extendLine
                            boxLeft = box.get_left(s.zoneBox)
                            line.set_x2(s.bb50Line, math.max(boxLeft, bar_index - 5))
                    if array.size(s.fvgBoxes) > 0
                        for fi = 0 to array.size(s.fvgBoxes) - 1
                            curFvgBox = array.get(s.fvgBoxes, fi)
                            box.set_bgcolor(curFvgBox,     color.new(#9e9e9e, 94))
                            box.set_border_color(curFvgBox, color.new(#9e9e9e, 100))
                            extendFVG = s.isBull ? i_extendFVGBull : i_extendFVGBear
                            if extendFVG
                                box.set_right(curFvgBox, bar_index)
                                if array.size(s.fvgCeLines) > fi
                                    line.set_x2(array.get(s.fvgCeLines, fi), bar_index)
                    if i_alertEnable and i_alertHit and not s.alertedInvalHit
                        alert("❌ Unicorn Model° Invalidation Hit | " + syminfo.ticker + " | " + (isBull ? "BULL" : "BEAR") + " | SL: " + str.tostring(s.invalidLevel, format.mintick), alert.freq_once_per_bar)
                        s.alertedInvalHit := true
            if (s.state == 1 or s.state == 2)
                retLabel := s.sweepLabel
    retLabel

    while array.size(setups) > i_history
        oldest = array.shift(setups)
        box.delete(oldest.zoneBox)
        for bx in oldest.fvgBoxes
            box.delete(bx)
        line.delete(oldest.sweepLine)
        label.delete(oldest.sweepLbl)
        line.delete(oldest.bb50Line)
        for cel in oldest.fvgCeLines
            line.delete(cel)
        for tl in oldest.targetLines
            line.delete(tl)
        for lbl in oldest.targetLabels
            label.delete(lbl)
    if i_discardInval

        if n > 0
            for i = n - 1 to 0
                s = array.get(setups, i)
                if s.state == 4
                    box.delete(s.zoneBox)
